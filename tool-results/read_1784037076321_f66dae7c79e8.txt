   853→/**
   854→ * Format context for PREMIUM users — full context with continuity and evolution signals.
   855→ *
   856→ * Context ordering follows cognitive priority for LLM consumption:
   857→ *
   858→ *   LAYER 1 — IDENTITY: Who this person is
   859→ *     Name + Onboarding origin (goals, focus, initial state)
   860→ *

   869→ *

  1017→          const hasB = pair[1].some(k => esText.includes(k));

  1297→      return `${h.name}: racha ${h.streak} días${status ? ` (${status})` : ''}`;
  1298→    });
  1299→    lines.push(`Hábitos activos: ${habitParts.join('. ')}.`);
  1300→  }
  1301→
  1302→  // Recent meditations
  1303→  if (ctx.recentMeditations.length > 0) {
  1304→    const last = ctx.recentMeditations[0];
  1305→    const days = daysAgo(last.completedAt);
  1306→    const when = days === 0 ? 'hoy' : days === 1 ? 'ayer' : `hace ${days} días`;
  1307→    lines.push(`Última meditación: ${last.duration} min de ${last.type.replace('_', ' ')} ${when}. ${ctx.recentMeditations.length} sesiones recientes.`);
  1308→  }
  1309→
  1310→  // Weekly activity summary + consistency signal
  1311→  const wa = ctx.weeklyActivity;
  1312→  const totalActivity = wa.meditations + wa.habits + wa.journals + wa.checkins + wa.wellness + wa.nutrition;
  1313→  if (totalActivity > 0) {
  1314→    const parts: string[] = [];
  1315→    if (wa.meditations > 0) parts.push(`${wa.meditations} meditación${wa.meditations > 1 ? 'es' : ''}`);
  1316→    if (wa.habits > 0) parts.push(`${wa.habits} hábito${wa.habits > 1 ? 's' : ''}`);
  1317→    if (wa.journals > 0) parts.push(`${wa.journals} entrada${wa.journals > 1 ? 's' : ''} de diario`);
  1318→    if (wa.checkins > 0) parts.push(`${wa.checkins} check-in${wa.checkins > 1 ? 's' : ''}`);
  1319→    if (wa.wellness > 0) parts.push(`${wa.wellness} registro${wa.wellness > 1 ? 's' : ''} de bienestar`);
  1320→    if (wa.nutrition > 0) parts.push(`${wa.nutrition} registro${wa.nutrition > 1 ? 's' : ''} de nutrición`);
  1321→    lines.push(`Esta semana: ${parts.join(', ')}.`);
  1322→  }
  1323→
  1324→  // Consistency evolution signal — compact, natural
  1325→  // Skip when ESE is active — its summary already provides the authoritative signal.
  1326→  const c = ctx.consistency;
  1327→  if (!es && c.trend === 'improving') {
  1328→    lines.push(`Viene siendo más constante últimamente.`);
  1329→  } else if (!es && c.trend === 'declining') {
  1330→    lines.push(`Últimamente menos activo/a que la semana pasada.`);
  1331→  } else if (c.trend === 'starting' && totalActivity <= 2) {
  1332→    lines.push(`Está empezando a usar la app.`);
  1333→  }
  1334→
  1335→  // Empire progress summary
  1336→  const activeEmpires = ctx.empireProgress.filter(e => e.xp > 0 || e.streak > 0);
  1337→  if (activeEmpires.length > 0) {
  1338→    // Find the empire with most XP to signal focus area
  1339→    const topEmpire = activeEmpires.reduce((a, b) => a.xp > b.xp ? a : b);
  1340→    const topEmpireName = EMPIRE_NAMES[topEmpire.empire] || topEmpire.empire;
  1341→
  1342→    const empireParts = activeEmpires.map(e => {
  1343→      const name = EMPIRE_NAMES[e.empire] || e.empire;
  1344→      return `${name}: nivel ${e.level}, ${e.xp} XP${e.streak > 0 ? `, racha ${e.streak} días` : ''}`;
  1345→    });
  1346→    lines.push(`Progreso de imperios: ${empireParts.join('. ')}. Enfocado/a especialmente en ${topEmpireName}.`);
  1347→  }
  1348→
  1349→  // ═══════════════════════════════════════════
  1350→  // LAYER 5 — CONVERSATIONAL MEMORY
  1351→  // ═══════════════════════════════════════════
  1352→
  1353→  // Recent conversation topics
  1354→  // Last layer: continuity between sessions matters, but the model should
  1355→  // prioritize current state over past conversations when they conflict.
  1356→  const namedConversations = ctx.recentConversations.filter(c => c.title !== 'Nueva conversación');
  1357→  if (namedConversations.length > 0) {
  1358→    const topics = namedConversations.map(c => {
  1359→      const days = daysAgo(c.updatedAt);
  1360→      const when = days === 0 ? 'hoy' : days === 1 ? 'ayer' : `hace ${days} días`;
  1361→      return `"${c.title}" (${when})`;
  1362→    });
  1363→    lines.push(`Temas recientes de conversación: ${topics.join(', ')}.`);
  1364→  }
  1365→
  1366→  return lines.join('\n');
  1367→}
