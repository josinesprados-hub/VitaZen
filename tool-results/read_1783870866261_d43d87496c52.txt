   105→interface ParsedCapture {
   106→  description: string;
   107→  amount: number;
   108→  category: string;
   109→  type: 'expense' | 'income';
   110→}
   111→
   112→function parseQuickCapture(input: string, historyMap: Record<string, string>): ParsedCapture | null {
   113→  if (!input.trim()) return null;
   114→
   115→  const tokens = input.trim().split(/\s+/);
   116→  let amount = 0;
   117→  let descriptionTokens: string[] = [];
   118→  let amountFound = false;
   119→
   120→  for (let i = tokens.length - 1; i >= 0; i--) {
   121→    if (/^[\d.,]+$/.test(tokens[i])) {
   122→      const parsed = parseEuropeanQuick(tokens[i]);
   123→      if (parsed > 0) {
   124→        amount = parsed;
   125→        descriptionTokens = tokens.slice(0, i);
   126→        amountFound = true;
   127→        break;
   128→      }
   129→    }
   130→  }
   131→
   132→  if (!amountFound && /^[\d.,]+$/.test(tokens[0])) {
   133→    const parsed = parseEuropeanQuick(tokens[0]);
   134→    if (parsed > 0) {
   135→      amount = parsed;
   136→      descriptionTokens = tokens.slice(1);
   137→      amountFound = true;
   138→    }
   139→  }
   140→
   141→  if (!amountFound) {
   142→    const match = input.match(/([\d.,]+)\s*€?\s*$/);
   143→    if (match) {
   144→      const parsed = parseEuropeanQuick(match[1]);
   145→      if (parsed > 0) {
   146→        amount = parsed;
   147→        descriptionTokens = [input.replace(match[0], '').trim()];
   148→        amountFound = true;
   149→      }
   150→    }
   151→  }
   152→
   153→  if (!amountFound || amount <= 0) return null;
   154→
   155→  const description = descriptionTokens.join(' ').trim();
   156→  const textToMatch = description || input;
   157→
   158→  let category = '';
   159→  const descLower = textToMatch.toLowerCase();
   160→  if (historyMap[descLower]) {
   161→    category = historyMap[descLower];
   162→  }
   163→
   164→  if (!category) {
   165→    for (const [regex, cat] of KEYWORD_CATEGORIES) {
   166→      if (regex.test(textToMatch)) {
   167→        category = cat;
   168→        break;
   169→      }
   170→    }
   171→  }
   172→
   173→  if (!category) category = 'Otros';
   174→
   175→  const type: 'expense' | 'income' = INCOME_KEYWORDS.test(textToMatch) ? 'income' : 'expense';
   176→
   177→  return {
   178→    description: description || category,
   179→    amount,
   180→    category,
   181→    type,
   182→  };
   183→}
   184→
   185→function parseEuropeanQuick(raw: string): number {
   186→  const trimmed = raw.trim();
   187→  if (!trimmed) return 0;
   188→  const hasComma = trimmed.includes(',');
   189→  const hasDot = trimmed.includes('.');
   190→  let normalized: string;
   191→
   192→  if (hasComma && hasDot) {
   193→    normalized = trimmed.replace(/\./g, '').replace(',', '.');
   194→  } else if (hasComma) {
   195→    normalized = trimmed.replace(',', '.');
   196→  } else if (hasDot) {
   197→    normalized = /^\d{1,3}(\.\d{3})+$/.test(trimmed)
   198→      ? trimmed.replace(/\./g, '')
   199→      : trimmed;
   200→  } else {
   201→    normalized = trimmed;
   202→  }
   203→
   204→  normalized = normalized.replace(/\.$/, '');
   205→  const parsed = parseFloat(normalized);
   206→  return isNaN(parsed) ? 0 : parsed;
   207→}

   213→function getMonthRange(date: Date) {
   214→  const start = new Date(date.getFullYear(), date.getMonth(), 1);
   215→  const end = new Date(date.getFullYear(), date.getMonth() + 1, 0, 23, 59, 59, 999);
   216→  return { start, end };
   217→}
   218→
   219→function filterLogsByRange(logs: FinanceLog[], start: Date, end: Date) {
   220→  return logs.filter(l => {
   221→    const d = new Date(l.date);
   222→    return d >= start && d <= end;
   223→  });
   224→}
   225→
   226→const MONTH_NAMES_ES = ['ene', 'feb', 'mar', 'abr', 'may', 'jun', 'jul', 'ago', 'sep', 'oct', 'nov', 'dic'];
   227→const MONTH_LABELS_ES = ['enero', 'febrero', 'marzo', 'abril', 'mayo', 'junio', 'julio', 'agosto', 'septiembre', 'octubre', 'noviembre', 'diciembre'];
   228→const DAY_NAMES_ES = ['dom', 'lun', 'mar', 'mié', 'jue', 'vie', 'sáb'];
   229→
   230→function formatDateLabel(dateStr: string): string {
   231→  const d = new Date(dateStr);
   232→  const day = DAY_NAMES_ES[d.getDay()];
   233→  const date = d.getDate();
   234→  const month = MONTH_NAMES_ES[d.getMonth()];
   235→  return `${day}, ${date} ${month}`;
   236→}

   238→function groupLogsByDate(logs: FinanceLog[]) {
   239→  const groups: Record<string, FinanceLog[]> = {};
   240→  for (const log of logs) {
   241→    const key = formatDateLabel(log.date);
   242→    if (!groups[key]) groups[key] = [];
   243→    groups[key].push(log);
   244→  }
   245→  return groups;
   246→}

   319→function IntentionBalance({ flows, totalExpense }: { flows: IntentionFlow[]; totalExpense: number }) {

   371→function QuickCapture({
   372→  onCapture,
   373→  onFullForm,
   374→  submitting,
   375→  historyMap,
   376→}: {
   377→  onCapture: (data: { type: string; category: string; amount: number; description: string; contexto: string }) => void;
   378→  onFullForm: () => void;
   379→  submitting: boolean;
   380→  historyMap: Record<string, string>;
   381→}) {
   382→  const [input, setInput] = useState('');
   383→  const [contexto, setContexto] = useState('');
   384→  const parsed = useMemo(() => parseQuickCapture(input, historyMap), [input, historyMap]);
   385→
   386→  const handleSubmit = () => {
   387→    if (!parsed || parsed.amount <= 0 || submitting) return;
   388→    onCapture({
   389→      type: parsed.type,
   390→      category: parsed.category,
   391→      amount: parsed.amount,
   392→      description: parsed.description,
   393→      contexto: contexto.trim(),
   394→    });
   395→    setInput('');
   396→    setContexto('');
   397→  };
   398→
   399→  const handleKeyDown = (e: React.KeyboardEvent) => {
   400→    if (e.key === 'Enter' && parsed && parsed.amount > 0) {
   401→      e.preventDefault();
   402→      handleSubmit();
   403→    }
   404→  };
   405→
   406→  return (
   407→    <div>
   408→      <div className="relative">
   409→        <input
   410→          type="text"
   411→          value={input}
   412→          onChange={(e) => setInput(e.target.value)}
   413→          onKeyDown={handleKeyDown}
   414→          placeholder="Café 3,50 · Taxi 14 · Nómina 1250"
   415→          className="w-full bg-[#000000] border border-[#1a1a1a] rounded-xl px-4 py-3.5 text-white text-base placeholder-[#444] focus:outline-none focus:border-champagne/40 transition-colors"
   416→          autoFocus
   417→          inputMode="text"
   418→          autoComplete="off"
   419→        />
   420→      </div>
   421→
   422→      {parsed && parsed.amount > 0 ? (
   423→        <div className="mt-2 flex items-center justify-between px-0.5">
   424→          <div className="flex items-center gap-1.5">
   425→            <span className={`text-[11px] font-medium ${parsed.type === 'income' ? 'text-champagne' : 'text-red-400/70'}`}>
   426→              {parsed.type === 'income' ? 'Ingreso' : 'Gasto'}
   427→            </span>
   428→            <span className="text-[#222] text-[11px]">·</span>
   429→            <span className="text-[11px] text-[#666]">{parsed.category}</span>
   430→            <span className="text-[#222] text-[11px]">·</span>
   431→            <span className={`text-[11px] font-medium tabular-nums ${parsed.type === 'income' ? 'text-champagne' : 'text-red-400/70'}`}>
   432→              {parsed.type === 'income' ? '+' : '-'}{formatCurrency(parsed.amount)}
   433→            </span>
   434→          </div>
   435→          <button
   436→            onClick={handleSubmit}
   437→            disabled={submitting}
   438→            className="bg-champagne text-black px-3 py-1 rounded-lg text-xs font-semibold hover:bg-champagne-hover active:scale-95 transition-all disabled:opacity-50"
   439→          >
   440→            {submitting ? '...' : 'OK'}
   441→          </button>
   442→        </div>
   443→      ) : (
   444→        <p className="mt-2 text-[10px] text-[#333] px-0.5">Escribe concepto y cantidad</p>
   445→      )}
   446→
   447→      <div className="mt-3">
   448→        <input
   449→          type="text"
   450→          value={contexto}
   451→          onChange={(e) => setContexto(e.target.value)}
   452→          placeholder="¿Qué pasó? (opcional)"
   453→          className="w-full bg-transparent border-b border-[#1a1a1a] px-0 py-1.5 text-[12px] text-[#888] placeholder-[#333] focus:outline-none focus:border-champagne/25 transition-colors italic"
   454→          autoComplete="off"
   455→        />
   456→      </div>
   457→
   458→      <button
   459→        onClick={onFullForm}
   460→        className="mt-2 text-[10px] text-[#444] hover:text-[#666] transition-colors tracking-wide"
   461→      >
   462→        Formulario completo
   463→      </button>
   464→    </div>
   465→  );
   466→}
