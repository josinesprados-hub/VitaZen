    67→function getRelativeDate(dateStr: string): string {
    68→  const date = new Date(dateStr);
    69→  const now = new Date();
    70→  const diffMs = now.getTime() - date.getTime();
    71→  const diffMins = Math.floor(diffMs / 60000);
    72→  const diffHours = Math.floor(diffMs / 3600000);
    73→
    74→  // Calendar-day difference using Madrid-normalized dates
    75→  const entryKey = getMadridDateKey(date);
    76→  const todayKey = getTodayDateKey();
    77→  const diffDays = daysBetweenDateKeys(entryKey, todayKey);
    78→
    79→  if (diffMins < 1 && diffDays === 0) return 'Ahora';
    80→  if (diffMins < 60 && diffDays === 0) return `Hace ${diffMins} min`;
    81→  if (diffHours < 24 && diffDays === 0) return `Hace ${diffHours}h`;
    82→  if (diffDays === 1) return 'Ayer';
    83→  if (diffDays < 7) return `Hace ${diffDays} días`;
    84→  if (diffDays < 30) {
    85→    const weeks = Math.floor(diffDays / 7);
    86→    return weeks === 1 ? 'Hace 1 semana' : `Hace ${weeks} semanas`;
    87→  }
    88→  const [eY, eM, eD] = entryKey.split('-').map(Number);
    89→  const entryDate = new Date(eY, eM - 1, eD);
    90→  return entryDate.toLocaleDateString('es-ES', { day: 'numeric', month: 'short' });
    91→}

    93→function getDateGroup(dateStr: string): string {
    94→  // Calendar-day difference using Madrid-normalized dates
    95→  const entryKey = getMadridDateKey(new Date(dateStr));
    96→  const todayKey = getTodayDateKey();
    97→
    98→  if (entryKey === todayKey) return 'Hoy';
    99→
   100→  const diffDays = daysBetweenDateKeys(entryKey, todayKey);
   101→
   102→  if (diffDays === 1) return 'Ayer';
   103→  if (diffDays < 7) return 'Esta semana';
   104→  if (diffDays < 30) return 'Este mes';
   105→  return 'Anterior';
   106→}

   125→function getProgressColor(rem: number, limit: number): string {
   126→  if (!isFinite(limit)) return '#c8a55a';
   127→  const pct = rem / limit;
   128→  if (pct > 0.5) return '#c8a55a';
   129→  if (pct > 0.25) return '#e8a849';
   130→  return '#ef4444';
   131→}

   519→  const sendMessage = async () => {
   520→    const content = input.trim();
   521→    if (!content || !activeThread || sendingRef.current) return;

   523→    const now = Date.now();
   524→    if (now - lastSendTime.current < 1000) return; // Debounce: 1s between sends
   525→    lastSendTime.current = now;
   526→
   527→    const sentThreadId = activeThread;
   528→    sendingRef.current = true;
   529→
   530→    const userMessage: Message = {
   531→      id: 'temp-' + Date.now(),
   532→      role: 'user',
   533→      content,
   534→      createdAt: new Date().toISOString(),
   535→    };
   536→    setMessages(prev => [...prev, userMessage]);
   537→    setInput('');
   538→    setSending(true);
   539→
   540→    try {
   541→      const res = await apiFetch('/api/ai/chat', {
   542→        method: 'POST',
   543→        body: JSON.stringify({ threadId: sentThreadId, content }),
   544→      });
   545→
   546→      if (res.status === 403) {
   547→        const data = await res.json();
   548→        setShowLimitModal(true);
   549→        setRemaining(0);
   550→        setDailyLimit(data.limit || 15);
   551→        // Remove the optimistic user message since it was blocked
   552→        setMessages(prev => prev.filter(m => m.id !== userMessage.id));
   553→        // L-4 FIX: Restore the input text so the user doesn't lose their message.
   554→        // Previously the input was cleared (line 547) and never restored on 403.
   555→        setInput(content);
   556→        return;
   557→      }
   558→
   559→      if (res.ok) {
   560→        const data = await res.json();
