   141→   141→
   142→   142→  // Previous week window for trend comparison
   143→   143→  const fourteenDaysAgo = new Date(now.getTime() - 14 * 86400000);
   144→   144→
   145→   145→  // Run all queries in parallel for performance

   178→   178→      where: { userId, streak: { gt: 0 } },
   179→   179→      orderBy: { streak: 'desc' },
   180→   180→      take: isPremium ? 8 : 4,
   181→   181→    }),
   182→   182→
   183→   183→    // Meditation sessions: FREE gets 1, PREMIUM gets 5

   198→   198→        ? { title: true, content: true, mood: true, createdAt: true }

   215→   215→      ? db.empireProgress.findMany({ where: { userId } })
   216→   216→      : Promise.resolve([]),
   217→   217→
   218→   218→    // Weekly meditation dates: PREMIUM only (for distinct-day counting)

   220→   220→      ? db.meditationSession.findMany({ where: { userId, completedAt: { gte: sevenDaysAgo } }, select: { completedAt: true } })

   230→   230→
   231→   231→    // Weekly habit log dates: last 7 days (for distinct-day counting)
   232→   232→    db.habitLog.findMany({
   233→   233→      where: { userId, lastCompletedAt: { gte: sevenDaysAgo } },
   234→   234→      select: { lastCompletedAt: true },
   235→   235→    }),
   236→   236→
   237→   237→    // Previous week checkins (for trend + emotional state): both tiers
   238→   238→    db.dailyCheckin.findMany({
   239→   239→      where: { userId, date: { gte: fourteenDaysAgo, lt: sevenDaysAgo } },

   248→   248→
   249→   249→    // Onboarding data: PREMIUM only

   255→   255→            primaryFocus: true,
   256→   256→            stressLevel: true,
   257→   257→            energyLevel: true,
   258→   258→            focusLevel: true,
   259→   259→            initialHabits: true,
   260→   260→          },
   261→   261→        })
   262→   262→      : Promise.resolve(null),
   263→   263→
   264→   264→    // Wellness logs: PREMIUM only — sleep quality + notes (dedup: mood/energy/stress already in DailyCheckin)
   265→   265→    // Extended to 90 days for PREMIUM: reuses this data for pattern detection (eliminates 6 redundant queries).
   266→   266→    // No take limit: full range needed for emotional state + pattern engine; display is sliced to 7 later.
   267→   267→    isPremium
   268→   268→      ? db.wellnessLog.findMany({
   269→   269→          where: { userId, date: { gte: ninetyDaysAgo } },
   270→   270→          orderBy: { date: 'desc' },
   271→   271→          select: { date: true, sleep: true, mood: true, energy: true, stress: true, notes: true },
   272→   272→        })
   273→   273→      : Promise.resolve([]),
   274→   274→
   275→   275→    // Finance logs: PREMIUM only — spending mood patterns + contexto + pattern engine input.
   276→   276→    // Extended to 90 days for PREMIUM: reuses this data for pattern detection (eliminates 6 redundant queries).
   277→   277→    isPremium
   278→   278→      ? db.financeLog.findMany({
   279→   279→          where: { userId, date: { gte: ninetyDaysAgo } },
   280→   280→          orderBy: { date: 'desc' },
   281→   281→          take: 200,
   282→   282→          select: { type: true, category: true, amount: true, mood: true, contexto: true, date: true },
   283→   283→        })
   284→   284→      : Promise.resolve([]),
   285→   285→
   286→   286→    // Weekly wellness log dates: PREMIUM only (for distinct-day counting)
   287→   287→    isPremium
   288→   288→      ? db.wellnessLog.findMany({ where: { userId, date: { gte: sevenDaysAgo } }, select: { date: true } })
   289→   289→      : Promise.resolve([] as { date: Date }[]),
   290→   290→
   291→   291→    // Weekly nutrition log dates: PREMIUM only (for distinct-day counting)
   292→   292→    isPremium
   293→   293→      ? db.nutritionLog.findMany({ where: { userId, date: { gte: sevenDaysAgo } }, select: { date: true } })
   294→   294→      : Promise.resolve([] as { date: Date }[]),
   295→   295→
   296→   296→    // Previous week habit logs: PREMIUM only (for emotional state prevWeekHabits)
   297→   297→    isPremium
   298→   298→      ? db.habitLog.findMany({ where: { userId, lastCompletedAt: { gte: fourteenDaysAgo, lt: sevenDaysAgo } }, select: { lastCompletedAt: true } })
   299→   299→      : Promise.resolve([] as { lastCompletedAt: Date | null }[]),
   300→   300→
   301→   301→    // Previous week meditations: PREMIUM only (for emotional state prevWeekMeditations)
   302→   302→    isPremium
   303→   303→      ? db.meditationSession.findMany({ where: { userId, completedAt: { gte: fourteenDaysAgo, lt: sevenDaysAgo } }, select: { completedAt: true } })
   304→   304→      : Promise.resolve([] as { completedAt: Date }[]),
   305→   305→
   306→   306→    // Previous week journals: PREMIUM only (for emotional state prevWeekJournals)
   307→   307→    isPremium
   308→   308→      ? db.journalEntry.findMany({ where: { userId, createdAt: { gte: fourteenDaysAgo, lt: sevenDaysAgo } }, select: { createdAt: true } })
   309→   309→      : Promise.resolve([] as { createdAt: Date }[]),
   310→   310→
   311→   311→    // Previous week nutrition logs: PREMIUM only (for emotional state prevWeekNutrition)
   312→   312→    isPremium
   313→   313→      ? db.nutritionLog.findMany({ where: { userId, date: { gte: fourteenDaysAgo, lt: sevenDaysAgo } }, select: { date: true } })
   314→   314→      : Promise.resolve([] as { date: Date }[]),
   315→   315→  ]);
   316→   316→
   317→   317→  // Derive counts from date arrays (for weeklyActivity display)
   318→   318→  const weeklyMeditationCount = weeklyMeditations.length;
   319→   319→  const weeklyJournalCount = weeklyJournals.length;
   320→   320→  const weeklyCheckinCount = weeklyCheckins.length;
   321→   321→  const weeklyHabitLogCount = weeklyHabitLogs.length;
   322→   322→  const weeklyWellnessCount = weeklyWellnessLogs.length;
   323→   323→  const weeklyNutritionCount = weeklyNutritionLogs.length;
   324→   324→
   325→   325→  // Weekly habit completions (PREMIUM only for display, but count for consistency)
   326→   326→  const weeklyHabits = isPremium
   327→   327→    ? habitStreaks.filter(h => {
   328→   328→        if (!h.lastCompletedAt) return false;
   329→   329→        return h.lastCompletedAt >= sevenDaysAgo;
   330→   330→      }).length
   331→   331→    : weeklyHabitLogCount;
   332→   332→
   333→   333→  // Compute consistency signal: count DISTINCT days with any activity
   334→   334→  const activeDaySet = new Set<string>();
   335→   335→  for (const c of weeklyCheckins) activeDaySet.add(toDateKey(c.date));

   338→   338→  }
   339→   339→  for (const m of weeklyMeditations) activeDaySet.add(toDateKey(m.completedAt));
   340→   340→  for (const j of weeklyJournals) activeDaySet.add(toDateKey(j.createdAt));
   341→   341→  for (const w of weeklyWellnessLogs) activeDaySet.add(toDateKey(w.date));
   342→   342→  for (const n of weeklyNutritionLogs) activeDaySet.add(toDateKey(n.date));
   343→   343→  const activeDaysThisWeek = Math.min(activeDaySet.size, 7);
   344→   344→
   345→   345→  const prevWeekCheckinCount = prevWeekCheckins.length;
   346→   346→
   347→   347→  // M-7: Compute trend based on aggregate activity, not just check-ins.
   348→   348→  // PREMIUM has full prev-week data → compare activeDaysThisWeek vs activeDaysPrevWeek.
   349→   349→  // FREE only has prev-week checkins → keep checkin-based trend (no other prev-week data available).
   350→   350→  let trend: 'improving' | 'stable' | 'declining' | 'starting' = 'stable';
   351→   351→  if (isPremium) {
   352→   352→    const prevActiveDaySet = new Set<string>();
   353→   353→    for (const c of prevWeekCheckins) prevActiveDaySet.add(toDateKey(c.date));
   354→   354→    for (const h of prevWeekHabitLogs) {

   356→   356→    }

   361→   361→    for (const n of prevWeekNutritionLogs) prevActiveDaySet.add(toDateKey(n.date));
   362→   362→    const activeDaysPrevWeek = Math.min(prevActiveDaySet.size, 7);
   363→   363→
   364→   364→    if (activeDaysThisWeek <= 1 && activeDaysPrevWeek <= 1) {
   365→   365→      trend = 'starting';
   366→   366→    } else if (activeDaysThisWeek > activeDaysPrevWeek + 1) {
   367→   367→      trend = 'improving';
   368→   368→    } else if (activeDaysThisWeek < activeDaysPrevWeek - 1) {
   369→   369→      trend = 'declining';
   370→   370→    }
   371→   371→  } else {
   372→   372→    if (weeklyCheckinCount <= 1 && prevWeekCheckinCount <= 1) {
   373→   373→      trend = 'starting';
   374→   374→    } else if (weeklyCheckinCount > prevWeekCheckinCount + 1) {
   375→   375→      trend = 'improving';
   376→   376→    } else if (weeklyCheckinCount < prevWeekCheckinCount - 1) {
   377→   377→      trend = 'declining';
   378→   378→    }
   379→   379→  }
   380→   380→

   382→   382→  let onboardingData: UserContext['onboardingData'] = null;
   383→   383→  if (onboardingRow) {
   384→   384→    let parsedGoals: string[] = [];
   385→   385→    try { parsedGoals = JSON.parse(onboardingRow.goals); } catch { /* keep empty */ }

   387→   387→    try { parsedHabits = JSON.parse(onboardingRow.initialHabits); } catch { /* keep empty */ }
   388→   388→    onboardingData = {

   391→   391→      stressLevel: onboardingRow.stressLevel,
   392→   392→      energyLevel: onboardingRow.energyLevel,
   393→   393→      focusLevel: onboardingRow.focusLevel,
   394→   394→      initialHabits: Array.isArray(parsedHabits) ? parsedHabits : [],

   396→   396→  }
   397→   397→
   398→   398→  // Fetch emotional state from the official engine (PREMIUM only; null for FREE)
   399→   399→  // This is the single source of truth — no recalculation, no duplication.
   400→   400→  // M-6: Reuse data already fetched in this Promise.all to avoid redundant gatherData() call.
   401→   401→  let emotionalState: UserContext['emotionalState'] = null;
   402→   402→  if (isPremium) {
   403→   403→    try {
   404→   404→      // Build RawData from already-fetched queries — no extra DB access
   405→   405→      const rawDataForEmotionalState: RawData = {
   406→   406→        thisWeekCheckins: weeklyCheckins,
   407→   407→        prevWeekCheckins,
   408→   408→        thisWeekHabits: weeklyHabitLogs,
   409→   409→        prevWeekHabits: prevWeekHabitLogs,
   410→   410→        allHabits: [],
   411→   411→        thisWeekMeditations: weeklyMeditations,
   412→   412→        prevWeekMeditations,
   413→   413→        thisWeekJournals: weeklyJournals,
   414→   414→        prevWeekJournals,
   415→   415→        thisWeekWellness: wellnessLogRows.filter(w => w.date >= sevenDaysAgo),
   416→   416→        prevWeekWellness: wellnessLogRows.filter(w => w.date < sevenDaysAgo),
   417→   417→        thisWeekNutrition: weeklyNutritionLogs,
   418→   418→        prevWeekNutrition: prevWeekNutritionLogs,
   419→   419→        thisWeekFinance: [],
   420→   420→        prevWeekFinance: [],
   421→   421→        empireProgress: [],
   422→   422→      };
   423→   423→
   424→   424→      const es = await getEmotionalState(userId, plan, rawDataForEmotionalState);
   425→   425→      emotionalState = {
   426→   426→        status: es.status,
   427→   427→        statusLabel: es.statusLabel,
   428→   428→        statusDescription: es.statusDescription,
   429→   429→        summary: es.summary,
   430→   430→        recommendation: es.recommendation,
   431→   431→      };
   432→   432→    } catch {
   433→   433→      // Non-blocking: if emotional state fails, continue without it
   434→   434→    }
   435→   435→  }
   436→   436→
   437→   437→  // Monthly closure records: consume existing closure records + their connections (PREMIUM only; empty for FREE)
   438→   438→  // The connections come from generateMonthlyDigest which already runs detectPatterns() for that month.
   439→   439→  // This gives the mentor awareness of what the user saw in their last closure — no recalculation.
   440→   440→  let recentClosures: {
   441→   441→    month: string;

   738→   738→function daysAgo(date: Date): number {
   739→   739→  const todayKey = getTodayDateKey();
   740→   740→  const dateKey = getMadridDateKey(date);

   742→   742→  const dateMs = new Date(dateKey + 'T00:00:00').getTime();
   743→   743→  return Math.round((todayMs - dateMs) / 86400000);
   744→   744→}

   751→   751→/**
   752→   752→ * Extract a fragment from journal content for the mentor context.
   753→   753→ * Returns null if the content doesn't add information beyond the title.
   754→   754→ *
   755→   755→ * Rules:
   756→   756→ * - Max 180 characters
   757→   757→ * - Never cut words
   758→   758→ * - Remove line breaks and repeated spaces
   759→   759→ * - End with "…" when truncated
   760→   760→ * - No summarization, no reinterpretation — exact user text only
   761→   761→ * - Skip if content only repeats what the title already says
   762→   762→ */
   763→   763→function extractJournalFragment(title: string, content: string): string | null {
   764→   764→  // Clean content: remove line breaks, collapse repeated spaces, trim
   765→   765→  const cleaned = content
   766→   766→    .replace(/\n/g, ' ')
   767→   767→    .replace(/\s+/g, ' ')
   768→   768→    .trim();
   769→   769→
   770→   770→  if (!cleaned) return null;
   771→   771→
   772→   772→  // Normalize title for comparison (lowercase, no punctuation, collapsed spaces)
   773→   773→  const normalizeForComparison = (s: string) =>
   774→   774→    s.toLowerCase().replace(/[^\wáéíóúñü]/g, ' ').replace(/\s+/g, ' ').trim();
   775→   775→
   776→   776→  const normalizedTitle = normalizeForComparison(title);
   777→   777→  const normalizedCleaned = normalizeForComparison(cleaned);
   778→   778→
   779→   779→  // If content starts with the title (user often repeats title as first line),
   780→   780→  // strip the title prefix from content for comparison
   781→   781→  let contentForComparison = normalizedCleaned;
   782→   782→  if (normalizedCleaned.startsWith(normalizedTitle)) {
   783→   783→    contentForComparison = normalizedCleaned.slice(normalizedTitle.length).trim();
   784→   784→  }
   785→   785→
   786→   786→  // If what remains is empty or too short, content doesn't add information
   787→   787→  if (contentForComparison.length < 10) return null;
   788→   788→
   789→   789→  // Truncate to 180 chars without cutting words
   790→   790→  const MAX_LEN = 180;
   791→   791→  if (cleaned.length <= MAX_LEN) return cleaned;
   792→   792→
   793→   793→  // Find the last space before the limit
   794→   794→  const truncated = cleaned.slice(0, MAX_LEN);
   795→   795→  const lastSpace = truncated.lastIndexOf(' ');
   796→   796→  if (lastSpace <= 0) return cleaned.slice(0, MAX_LEN) + '…';
   797→   797→
   798→   798→  return truncated.slice(0, lastSpace) + '…';
   799→   799→}

  1249→  1249→    const monthsAgo = (nowYear - cYear) * 12 + (nowMonth - cMonth);
  1250→  1250→
  1251→  1251→    const reflectionsCount = closures.filter(cl => cl.hasReflection).length;
  1252→  1252→
  1253→  1253→    if (monthsAgo <= 1) {
  1254→  1254→      // Recent closure — user is engaged
  1255→  1255→      if (reflectionsCount >= 2) {
  1256→  1256→        lines.push(`Viene realizando cierres personales con regularidad.`);
  1257→  1257→      } else if (latest.hasReflection) {
  1258→  1258→        lines.push(`Su última reflexión mensual fue reciente.`);
  1259→  1259→      } else {
  1260→  1260→        lines.push(`Revisó su cierre mensual recientemente.`);
  1261→  1261→      }
  1262→  1262→    } else if (monthsAgo <= 3) {
  1263→  1263→      // Not too long ago
  1264→  1264→      if (reflectionsCount >= 2) {
  1265→  1265→        lines.push(`Suele reflexionar sobre su evolución cada mes.`);
  1266→  1266→      } else {
  1267→  1267→        lines.push(`Hace poco revisó cómo le fue en un mes.`);
  1268→  1268→      }
  1269→  1269→    } else {
  1270→  1270→      // A while since last closure
  1271→  1271→      lines.push(`Hace tiempo que no realiza un cierre mensual.`);
  1272→  1272→    }
  1273→  1273→
  1274→  1274→    // Connections from the closure digest — only if the closure is recent (≤3 months)
  1275→  1275→    // and the connection hasn't already been mentioned in pattern observations.
  1276→  1276→    if (monthsAgo <= 3 && latest.connections.length > 0) {
  1277→  1277→      const patternIds = new Set(
  1278→  1278→        (ctx.patternObservations || []).map(p => p.id)
  1279→  1279→      );
  1280→  1280→      const closureConns = latest.connections.filter(c => !patternIds.has(c.id));
  1281→  1281→      if (closureConns.length > 0) {
  1282→  1282→        const connTexts = closureConns.map(c => c.text).join('. ');
  1283→  1283→        lines.push(`En su cierre mensual, VitaZen también observó: ${connTexts}.`);
  1284→  1284→      }
  1285→  1285→    }
  1286→  1286→  }
  1287→  1287→
  1288→  1288→  // ═══════════════════════════════════════════
  1289→  1289→  // LAYER 4 — BEHAVIORAL PATTERNS
  1290→  1290→  // ═══════════════════════════════════════════
  1291→  1291→
  1292→  1292→  // Habit streaks with detail
  1293→  1293→  if (ctx.habitStreaks.length > 0) {
  1294→  1294→    const habitParts = ctx.habitStreaks.slice(0, 5).map(h => {
  1295→  1295→      const daysSince = h.lastCompletedAt ? daysAgo(h.lastCompletedAt) : null;
  1296→  1296→      const status = daysSince === 0 ? 'hecho hoy' : daysSince === 1 ? 'hecho ayer' : daysSince !== null ? `último hace ${daysSince} días` : '';
  1297→  1297→      return `${h.name}: racha ${h.streak} días${status ? ` (${status})` : ''}`;
  1298→  1298→    });
  1299→  1299→    lines.push(`Hábitos activos: ${habitParts.join('. ')}.`);
  1300→  1300→  }
  1301→  1301→
  1302→  1302→  // Recent meditations
  1303→  1303→  if (ctx.recentMeditations.length > 0) {
  1304→  1304→    const last = ctx.recentMeditations[0];
  1305→  1305→    const days = daysAgo(last.completedAt);
  1306→  1306→    const when = days === 0 ? 'hoy' : days === 1 ? 'ayer' : `hace ${days} días`;
  1307→  1307→    lines.push(`Última meditación: ${last.duration} min de ${last.type.replace('_', ' ')} ${when}. ${ctx.recentMeditations.length} sesiones recientes.`);
  1308→  1308→  }
  1309→  1309→
  1310→  1310→  // Weekly activity summary + consistency signal
  1311→  1311→  const wa = ctx.weeklyActivity;
  1312→  1312→  const totalActivity = wa.meditations + wa.habits + wa.journals + wa.checkins + wa.wellness + wa.nutrition;
  1313→  1313→  if (totalActivity > 0) {
  1314→  1314→    const parts: string[] = [];
  1315→  1315→    if (wa.meditations > 0) parts.push(`${wa.meditations} meditación${wa.meditations > 1 ? 'es' : ''}`);
  1316→  1316→    if (wa.habits > 0) parts.push(`${wa.habits} hábito${wa.habits > 1 ? 's' : ''}`);
  1317→  1317→    if (wa.journals > 0) parts.push(`${wa.journals} entrada${wa.journals > 1 ? 's' : ''} de diario`);
  1318→  1318→    if (wa.checkins > 0) parts.push(`${wa.checkins} check-in${wa.checkins > 1 ? 's' : ''}`);
  1319→  1319→    if (wa.wellness > 0) parts.push(`${wa.wellness} registro${wa.wellness > 1 ? 's' : ''} de bienestar`);
  1320→  1320→    if (wa.nutrition > 0) parts.push(`${wa.nutrition} registro${wa.nutrition > 1 ? 's' : ''} de nutrición`);
  1321→  1321→    lines.push(`Esta semana: ${parts.join(', ')}.`);
  1322→  1322→  }
  1323→  1323→
  1324→  1324→  // Consistency evolution signal — compact, natural
  1325→  1325→  // Skip when ESE is active — its summary already provides the authoritative signal.
  1326→  1326→  const c = ctx.consistency;
  1327→  1327→  if (!es && c.trend === 'improving') {
  1328→  1328→    lines.push(`Viene siendo más constante últimamente.`);
  1329→  1329→  } else if (!es && c.trend === 'declining') {
  1330→  1330→    lines.push(`Últimamente menos activo/a que la semana pasada.`);
  1331→  1331→  } else if (c.trend === 'starting' && totalActivity <= 2) {
  1332→  1332→    lines.push(`Está empezando a usar la app.`);
  1333→  1333→  }
  1334→  1334→
  1335→  1335→  // Empire progress summary
  1336→  1336→  const activeEmpires = ctx.empireProgress.filter(e => e.xp > 0 || e.streak > 0);
  1337→  1337→  if (activeEmpires.length > 0) {
  1338→  1338→    // Find the empire with most XP to signal focus area
  1339→  1339→    const topEmpire = activeEmpires.reduce((a, b) => a.xp > b.xp ? a : b);
  1340→  1340→    const topEmpireName = EMPIRE_NAMES[topEmpire.empire] || topEmpire.empire;
  1341→  1341→
  1342→  1342→    const empireParts = activeEmpires.map(e => {
  1343→  1343→      const name = EMPIRE_NAMES[e.empire] || e.empire;
  1344→  1344→      return `${name}: nivel ${e.level}, ${e.xp} XP${e.streak > 0 ? `, racha ${e.streak} días` : ''}`;
  1345→  1345→    });
  1346→  1346→    lines.push(`Progreso de imperios: ${empireParts.join('. ')}. Enfocado/a especialmente en ${topEmpireName}.`);
  1347→  1347→  }
  1348→  1348→
  1349→  1349→  // ═══════════════════════════════════════════
  1350→  1350→  // LAYER 5 — CONVERSATIONAL MEMORY
  1351→  1351→  // ═══════════════════════════════════════════
  1352→  1352→
  1353→  1353→  // Recent conversation topics
  1354→  1354→  // Last layer: continuity between sessions matters, but the model should
  1355→  1355→  // prioritize current state over past conversations when they conflict.
  1356→  1356→  const namedConversations = ctx.recentConversations.filter(c => c.title !== 'Nueva conversación');
  1357→  1357→  if (namedConversations.length > 0) {
  1358→  1358→    const topics = namedConversations.map(c => {
  1359→  1359→      const days = daysAgo(c.updatedAt);
  1360→  1360→      const when = days === 0 ? 'hoy' : days === 1 ? 'ayer' : `hace ${days} días`;
  1361→  1361→      return `"${c.title}" (${when})`;
  1362→  1362→    });
  1363→  1363→    lines.push(`Temas recientes de conversación: ${topics.join(', ')}.`);
  1364→  1364→  }
  1365→  1365→
  1366→  1366→  return lines.join('\n');
  1367→  1367→}
