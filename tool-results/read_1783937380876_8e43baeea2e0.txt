    73→   112→   112→function parseQuickCapture(input: string, historyMap: Record<string, string>): ParsedCapture | null {
    74→   113→   113→  if (!input.trim()) return null;
    75→   114→   114→
    76→   115→   115→  const tokens = input.trim().split(/\s+/);
    77→   116→   116→  let amount = 0;
    78→   117→   117→  let descriptionTokens: string[] = [];
    79→   118→   118→  let amountFound = false;
    80→   119→   119→
    81→   120→   120→  for (let i = tokens.length - 1; i >= 0; i--) {
    82→   121→   121→    if (/^[\d.,]+$/.test(tokens[i])) {
    83→   122→   122→      const parsed = parseEuropeanQuick(tokens[i]);
    84→   123→   123→      if (parsed > 0) {
    85→   124→   124→        amount = parsed;
    86→   125→   125→        descriptionTokens = tokens.slice(0, i);
    87→   126→   126→        amountFound = true;
    88→   127→   127→        break;
    89→   128→   128→      }
    90→   129→   129→    }
    91→   130→   130→  }
    92→   131→   131→
    93→   132→   132→  if (!amountFound && /^[\d.,]+$/.test(tokens[0])) {
    94→   133→   133→    const parsed = parseEuropeanQuick(tokens[0]);
    95→   134→   134→    if (parsed > 0) {
    96→   135→   135→      amount = parsed;
    97→   136→   136→      descriptionTokens = tokens.slice(1);
    98→   137→   137→      amountFound = true;
    99→   138→   138→    }
   100→   139→   139→  }
   101→   140→   140→
   102→   141→   141→  if (!amountFound) {
   103→   142→   142→    const match = input.match(/([\d.,]+)\s*€?\s*$/);
   104→   143→   143→    if (match) {
   105→   144→   144→      const parsed = parseEuropeanQuick(match[1]);
   106→   145→   145→      if (parsed > 0) {
   107→   146→   146→        amount = parsed;
   108→   147→   147→        descriptionTokens = [input.replace(match[0], '').trim()];
   109→   148→   148→        amountFound = true;
   110→   149→   149→      }
   111→   150→   150→    }
   112→   151→   151→  }
   113→   152→   152→
   114→   153→   153→  if (!amountFound || amount <= 0) return null;
   115→   154→   154→
   116→   155→   155→  const description = descriptionTokens.join(' ').trim();
   117→   156→   156→  const textToMatch = description || input;
   118→   157→   157→
   119→   158→   158→  let category = '';
   120→   159→   159→  const descLower = textToMatch.toLowerCase();
   121→   160→   160→  if (historyMap[descLower]) {
   122→   161→   161→    category = historyMap[descLower];
   123→   162→   162→  }
   124→   163→   163→
   125→   164→   164→  if (!category) {
   126→   165→   165→    for (const [regex, cat] of KEYWORD_CATEGORIES) {
   127→   166→   166→      if (regex.test(textToMatch)) {
   128→   167→   167→        category = cat;
   129→   168→   168→        break;
   130→   169→   169→      }
   131→   170→   170→    }
   132→   171→   171→  }
   133→   172→   172→
   134→   173→   173→  if (!category) category = 'Otros';
   135→   174→   174→
   136→   175→   175→  const type: 'expense' | 'income' = INCOME_KEYWORDS.test(textToMatch) ? 'income' : 'expense';
   137→   176→   176→
   138→   177→   177→  return {
   139→   178→   178→    description: description || category,
   140→   179→   179→    amount,
   141→   180→   180→    category,
   142→   181→   181→    type,
   143→   182→   182→  };
   144→   183→   183→}
   145→   184→   184→
   146→   185→   185→function parseEuropeanQuick(raw: string): number {
   147→   186→   186→  const trimmed = raw.trim();
   148→   187→   187→  if (!trimmed) return 0;
   149→   188→   188→  const hasComma = trimmed.includes(',');
   150→   189→   189→  const hasDot = trimmed.includes('.');
   151→   190→   190→  let normalized: string;
   152→   191→   191→
   153→   192→   192→  if (hasComma && hasDot) {
   154→   193→   193→    normalized = trimmed.replace(/\./g, '').replace(',', '.');
   155→   194→   194→  } else if (hasComma) {
   156→   195→   195→    normalized = trimmed.replace(',', '.');
   157→   196→   196→  } else if (hasDot) {
   158→   197→   197→    normalized = /^\d{1,3}(\.\d{3})+$/.test(trimmed)
   159→   198→   198→      ? trimmed.replace(/\./g, '')
   160→   199→   199→      : trimmed;
   161→   200→   200→  } else {
   162→   201→   201→    normalized = trimmed;
   163→   202→   202→  }
   164→   203→   203→
   165→   204→   204→  normalized = normalized.replace(/\.$/, '');
   166→   205→   205→  const parsed = parseFloat(normalized);
   167→   206→   206→  return isNaN(parsed) ? 0 : parsed;
   168→   207→   207→}

   336→   375→   375→  historyMap,
