    87→function getScoreRingColor(score: number): string {
    88→  if (score >= 75) return '#22c55e';
    89→  if (score >= 50) return '#c8a55a';
    90→  if (score >= 30) return '#e8a849';
    91→  return '#ef4444';
    92→}
    93→
    94→function getMetricLevel(value: number): string {
    95→  if (value >= 65) return 'Alto';
    96→  if (value >= 35) return 'Medio';
    97→  return 'Bajo';
    98→}
    99→
   100→function getMetricColor(value: number): string {
   101→  if (value >= 65) return '#c8a55a';
   102→  if (value >= 35) return '#999';
   103→  return '#666';
   104→}
   105→
   106→function TrendBadge({ value, invert = false }: { value: number; invert?: boolean }) {
   107→  // invert=true means positive when value is negative (e.g. stress going down)
   108→  const positive = invert ? value < -0.2 : value > 0.2;
   109→  const negative = invert ? value > 0.2 : value < -0.2;
   110→
   111→  if (positive) {
   112→    return (
   113→      <span className="flex items-center gap-1 text-[11px] text-[#22c55e]">
   114→        <TrendingUp size={11} /> {invert ? '↓' : '↑'}
   115→      </span>
   116→    );
   117→  }
   118→  if (negative) {
   119→    return (
   120→      <span className="flex items-center gap-1 text-[11px] text-champagne-warm">
   121→        <TrendingDown size={11} /> {invert ? '↑' : '↓'}
   122→      </span>
   123→    );
   124→  }
   125→  return (
   126→    <span className="flex items-center gap-1 text-[11px] text-[#555]">
   127→      <Minus size={11} /> —
   128→    </span>
   129→  );
   130→}
   131→
   132→function getInsightBorderClass(type: string): string {
   133→  switch (type) {
   134→    case 'positive': return 'border-[#22c55e]/15 hover:border-[#22c55e]/30';
   135→    case 'warning': return 'border-champagne-warm/15 hover:border-champagne-warm/30';
   136→    case 'trend': return 'border-champagne/15 hover:border-champagne/30';
   137→    default: return 'border-[#1a1a1a] hover:border-[#2a2a2a]';
   138→  }
   139→}
