    67→    67→function getRelativeDate(dateStr: string): string {
    68→    68→  const date = new Date(dateStr);
    69→    69→  const now = new Date();
    70→    70→  const diffMs = now.getTime() - date.getTime();
    71→    71→  const diffMins = Math.floor(diffMs / 60000);
    72→    72→  const diffHours = Math.floor(diffMs / 3600000);
    73→    73→
    74→    74→  // Calendar-day difference using Madrid-normalized dates
    75→    75→  const entryKey = getMadridDateKey(date);
    76→    76→  const todayKey = getTodayDateKey();
    77→    77→  const [eY, eM, eD] = entryKey.split('-').map(Number);
    78→    78→  const [tY, tM, tD] = todayKey.split('-').map(Number);
    79→    79→  const entryDate = new Date(eY, eM - 1, eD);
    80→    80→  const todayDate = new Date(tY, tM - 1, tD);
    81→    81→  const diffDays = Math.round((todayDate.getTime() - entryDate.getTime()) / 86400000);
    82→    82→
    83→    83→  if (diffMins < 1 && diffDays === 0) return 'Ahora';
    84→    84→  if (diffMins < 60 && diffDays === 0) return `Hace ${diffMins} min`;
    85→    85→  if (diffHours < 24 && diffDays === 0) return `Hace ${diffHours}h`;
    86→    86→  if (diffDays === 1) return 'Ayer';
    87→    87→  if (diffDays < 7) return `Hace ${diffDays} días`;
    88→    88→  if (diffDays < 30) {
    89→    89→    const weeks = Math.floor(diffDays / 7);
    90→    90→    return weeks === 1 ? 'Hace 1 semana' : `Hace ${weeks} semanas`;
    91→    91→  }
    92→    92→  return entryDate.toLocaleDateString('es-ES', { day: 'numeric', month: 'short' });
    93→    93→}

    95→    95→function getDateGroup(dateStr: string): string {
    96→    96→  // Calendar-day difference using Madrid-normalized dates
    97→    97→  const entryKey = getMadridDateKey(new Date(dateStr));
    98→    98→  const todayKey = getTodayDateKey();
    99→    99→
   100→   100→  if (entryKey === todayKey) return 'Hoy';
   101→   101→
   102→   102→  const [eY, eM, eD] = entryKey.split('-').map(Number);
   103→   103→  const [tY, tM, tD] = todayKey.split('-').map(Number);
   104→   104→  const entryDate = new Date(eY, eM - 1, eD);
   105→   105→  const todayDate = new Date(tY, tM - 1, tD);
   106→   106→  const diffDays = Math.round((todayDate.getTime() - entryDate.getTime()) / 86400000);
   107→   107→
   108→   108→  if (diffDays === 1) return 'Ayer';
   109→   109→  if (diffDays < 7) return 'Esta semana';
   110→   110→  if (diffDays < 30) return 'Este mes';
   111→   111→  return 'Anterior';
   112→   112→}
