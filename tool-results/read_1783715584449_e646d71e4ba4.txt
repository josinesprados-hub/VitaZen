    10→// ─── Error serialization for observability ──────────────────────────
    11→// Extracts detailed error information without altering the error
    12→// handling flow. Designed for Prisma, Stripe, and generic errors.
    13→// ONLY used for logging — never changes behavior.
    14→
    15→function serializeWebhookError(error: unknown): Record<string, unknown> {
    16→  const details: Record<string, unknown> = {};
    17→
    18→  // Constructor name + message + stack for Error instances
    19→  if (error instanceof Error) {
    20→    details.errorConstructor = error.constructor.name;
    21→    details.errorMessage = error.message;
    22→    details.errorStack = error.stack;
    23→  } else {
    24→    details.errorConstructor = typeof error;
    25→    try {
    26→      details.errorRaw = String(error);
    27→    } catch {
    28→      details.errorRaw = '[unstringifiable]';
    29→    }
    30→  }
    31→
    32→  // Prisma-specific: P2002 (unique constraint), P2025 (not found), P2003 (FK), etc.
    33→  if (error && typeof error === 'object' && 'code' in error) {
    34→    const prismaLike = error as { code?: string; meta?: unknown; clientVersion?: string };
    35→    if (typeof prismaLike.code === 'string' && /^P\d{4}$/.test(prismaLike.code)) {
    36→      details.prismaCode = prismaLike.code;
    37→      try {
    38→        details.prismaMeta = JSON.stringify(prismaLike.meta);
    39→      } catch {
    40→        details.prismaMeta = String(prismaLike.meta);
    41→      }
    42→      details.prismaClientVersion = prismaLike.clientVersion;
    43→    }
    44→  }
    45→
    46→  // Stripe-specific: Stripe.errors.* (identified by rawType property)
    47→  if (error && typeof error === 'object' && 'rawType' in error) {
    48→    const stripeLike = error as { type?: string; code?: string; requestId?: string; statusCode?: number };
    49→    details.stripeErrorType = stripeLike.type;
    50→    details.stripeErrorCode = stripeLike.code;
    51→    details.stripeRequestId = stripeLike.requestId;
    52→    details.stripeStatusCode = stripeLike.statusCode;
    53→  }
    54→
    55→  // Safe serialization fallback — captures all own properties
    56→  try {
    57→    details.errorSerialized = JSON.stringify(error, Object.getOwnPropertyNames(error));
    58→  } catch {
    59→    details.errorSerialized = '[not serializable]';
    60→  }
    61→
    62→  return details;
    63→}

    65→// ─── Idempotency: prevent duplicate event processing ─────────────────
    66→// Stripe retries webhooks on 500 responses or connection drops.
    67→// Without dedup, a single event could:
    68→//   - Create duplicate Subscription records
    69→//   - Send duplicate confirmation emails
    70→//   - Track duplicate analytics events
    71→// The StripeEventLog table stores processed event IDs.
    72→// Auto-cleaned: events older than 7 days are pruned on each webhook call.
    73→
    74→const EVENT_TTL_DAYS = 7;
    75→
    76→async function isEventProcessed(eventId: string): Promise<boolean> {
    77→  try {
    78→    const existing = await db.stripeEventLog.findUnique({ where: { eventId } });
    79→    return !!existing;
    80→  } catch {
    81→    // If the table doesn't exist yet (migration not applied), log and continue
    82→    // rather than blocking ALL webhook processing.
    83→    serverLog.warn('webhook/stripe', 'StripeEventLog table not available — skipping dedup', { eventId });
    84→    return false;
    85→  }
    86→}
    87→
    88→async function markEventProcessed(eventId: string, eventType: string): Promise<void> {
    89→  try {
    90→    await db.stripeEventLog.create({
    91→      data: { eventId, eventType },
    92→    });
    93→  } catch {
    94→    // Unique constraint violation = already processed (race condition)
    95→    // This is fine — another worker beat us to it.
    96→  }
    97→}

    99→async function cleanupOldEvents(): Promise<void> {
   100→  try {
   101→    const cutoff = new Date(Date.now() - EVENT_TTL_DAYS * 24 * 60 * 60 * 1000);
   102→    const result = await db.stripeEventLog.deleteMany({
   103→      where: { createdAt: { lt: cutoff } },
   104→    });
   105→    if (result.count > 0) {
   106→      serverLog.info('webhook/stripe', 'Cleaned up old event logs', { count: result.count });
   107→    }
   108→  } catch {
   109→    // Non-critical: cleanup failure must never block webhook processing
   110→  }
   111→}

   113→// ─── Resolve userId from checkout session ────────────────────────────
   114→// Primary: session.metadata.userId (set during checkout creation)
   115→// Fallback: lookup by session.customer → customer.metadata.userId
   116→// This handles edge cases where metadata wasn't set (e.g. manual
   117→// checkout created via Stripe Dashboard).
   118→
   119→async function resolveUserId(session: Stripe.Checkout.Session): Promise<string | null> {
   120→  // Primary path: userId in session metadata
   121→  if (session.metadata?.userId) {
   122→    return session.metadata.userId;
   123→  }
   124→
   125→  // Fallback: look up customer metadata
   126→  const customerId = session.customer as string | null;
   127→  if (!customerId) {
   128→    serverLog.warn('webhook/stripe', 'No userId in metadata and no customer ID — cannot resolve user');
   129→    return null;
   130→  }
   131→
   132→  try {
   133→    const customer = await stripe.customers.retrieve(customerId);
   134→    if ('deleted' in customer && customer.deleted) {
   135→      serverLog.warn('webhook/stripe', 'Customer deleted — cannot resolve userId', { customerId });
   136→      return null;
   137→    }
   138→    if ('metadata' in customer && customer.metadata?.userId) {
   139→      serverLog.info('webhook/stripe', 'Resolved userId from customer metadata (fallback)');
   140→      return customer.metadata.userId;
   141→    }
   142→  } catch (e) {
   143→    serverLog.error('webhook/stripe', 'Failed to retrieve customer for userId fallback', e);
   144→  }
   145→
   146→  // Last resort: search our DB by stripeCustomerId
   147→  try {
   148→    const user = await db.user.findUnique({ where: { stripeCustomerId: customerId } });
   149→    if (user) {
   150→      serverLog.info('webhook/stripe', 'Resolved userId from DB stripeCustomerId (last resort)');
   151→      return user.id;
   152→    }
   153→  } catch {
   154→    // DB lookup failed
   155→  }
   156→
   157→  serverLog.warn('webhook/stripe', 'Could not resolve userId for session');
   158→  return null;
   159→}

   161→export async function POST(request: NextRequest) {
   162→  const start = Date.now();
   163→  let eventType = 'unknown';
   164→  let eventId = 'unknown';
   165→  let checkoutSessionContext: Record<string, unknown> | null = null;
   166→
   167→  const body = await request.text();
   168→  const signature = request.headers.get('stripe-signature');
   169→
   170→  if (!signature) {
   171→    return NextResponse.json({ error: 'No signature' }, { status: 400 });
   172→  }
   173→
   174→  let event: Stripe.Event;
   175→
   176→  try {
   177→    event = stripe.webhooks.constructEvent(body, signature, process.env.STRIPE_WEBHOOK_SECRET!);
   178→    eventType = event.type;
   179→    eventId = event.id;
   180→  } catch (liveErr) {
   181→    // Live secret failed — try Test secret if available, but ONLY in non-production.
   182→    // BUG-B1 FIX: In production, accepting test-secret-signed events allows anyone
   183→    // with the (less protected) test secret to forge webhook events and grant
   184→    // themselves Premium. The test secret fallback is now gated behind
   185→    // NODE_ENV !== 'production' so it only works in local development.
   186→    const testSecret = process.env.STRIPE_TEST_WEBHOOK_SECRET;
   187→    const isProduction = process.env.NODE_ENV === 'production';
   188→    if (testSecret && !isProduction) {
   189→      try {
   190→        event = stripe.webhooks.constructEvent(body, signature, testSecret);
   191→        eventType = event.type;
   192→        eventId = event.id;
   193→        serverLog.info('webhook/stripe', 'Webhook verified with Test secret (non-production)', { eventType, eventId });
   194→      } catch (testErr) {
   195→        serverLog.error('webhook/stripe', 'Webhook signature verification failed (both secrets)', testErr, {
   196→          eventType,
   197→        });
   198→        return NextResponse.json({ error: 'Invalid signature' }, { status: 400 });
   199→      }
   200→    } else {
   201→      serverLog.error('webhook/stripe', 'Webhook signature verification failed', liveErr, {
   202→        eventType,
   203→      });
   204→      return NextResponse.json({ error: 'Invalid signature' }, { status: 400 });
   205→    }
   206→  }
   207→
   208→  // ─── Idempotency check ──────────────────────────────────────────
   209→  if (await isEventProcessed(event.id)) {
   210→    serverLog.info('webhook/stripe', 'Duplicate event skipped', { eventType, eventId });
   211→    return NextResponse.json({ received: true, deduplicated: true });
   212→  }
   213→
   214→  try {
   215→    switch (event.type) {
   216→      case 'checkout.session.completed': {
   217→        const session = event.data.object as Stripe.Checkout.Session;
   218→        const customerId = session.customer as string | null;
   219→        const userId = await resolveUserId(session);
   220→
   221→        serverLog.info('webhook/stripe', 'checkout.session.completed', {
   222→          userId: userId ? '[RESOLVED]' : 'null',
   223→          customerId: customerId || 'null',
   224→          eventId,
   225→        });
   226→
   227→        // Store context for error observability in outer catch
   228→        checkoutSessionContext = {
   229→          userId,
   230→          customerId: customerId || null,
   231→          subscriptionId: session.subscription ?? null,
   232→          sessionId: session.id,
   233→        };
   234→
   235→        if (!userId) {
   236→          serverLog.error('webhook/stripe', 'CANNOT resolve userId — customer paid but will NOT get premium. Manual intervention required.', undefined, {
   237→            sessionId: session.id,
   238→            eventType,
   239→          });
   240→          // Still mark as processed to avoid infinite retries
   241→          await markEventProcessed(event.id, event.type);
   242→          await cleanupOldEvents();
   243→          break;
   244→        }
   245→
   246→        // Fetch line items explicitly — they are NOT included in the event object
   247→        const lineItems = await stripe.checkout.sessions.listLineItems(session.id);
   248→        const stripePriceId = lineItems.data[0]?.price?.id || '';
   249→
   250→        // Get subscription period dates if available
   251→        // Stripe Basil/Dahlia: current_period_start/end moved from Subscription
   252→        // to SubscriptionItem. Fallback to defaults if unavailable.
   253→        const subscriptionId = session.subscription as string;
   254→        let periodStart = new Date();
   255→        let periodEnd = new Date(Date.now() + 30 * 24 * 60 * 60 * 1000);
   256→
   257→        if (subscriptionId) {
   258→          try {
   259→            const subscription = await stripe.subscriptions.retrieve(subscriptionId);
   260→            const firstItem = subscription.items.data[0];
   261→            if (firstItem && typeof firstItem.current_period_start === 'number' && typeof firstItem.current_period_end === 'number') {
   262→              periodStart = new Date(firstItem.current_period_start * 1000);
   263→              periodEnd = new Date(firstItem.current_period_end * 1000);
   264→            } else {
   265→              serverLog.error('webhook/stripe', 'Subscription item missing period dates — using defaults', undefined, {
   266→                subscriptionId,
   267→                hasItems: !!firstItem,
   268→                itemStartType: typeof firstItem?.current_period_start,
   269→                itemEndType: typeof firstItem?.current_period_end,
   270→              });
   271→            }
   272→          } catch (e) {
   273→            serverLog.error('webhook/stripe', 'Could not retrieve subscription details', e, {
   274→              subscriptionId,
   275→            });
   276→          }
   277→        }
   278→
   279→        // ─── ATOMIC: user update + subscription create in one transaction ───
   280→        await db.$transaction(async (tx) => {
   281→          // Mark any existing active subscriptions as superseded
   282→          const existingActive = await tx.subscription.findFirst({
   283→            where: { userId, status: 'active' },
   284→          });
   285→
   286→          if (existingActive) {
   287→            serverLog.warn('webhook/stripe', 'User already has active subscription — marking old as superseded', {
   288→              oldSubId: existingActive.id,
   289→            });
   290→            await tx.subscription.update({
   291→              where: { id: existingActive.id },
   292→              data: { status: 'superseded' },
   293→            });
   294→          }
   295→
   296→          // Update user plan + stripeCustomerId atomically
   297→          const updateData: Record<string, unknown> = { plan: 'PREMIUM' };
   298→          if (customerId) {
   299→            updateData.stripeCustomerId = customerId;
   300→          } else {
   301→            serverLog.warn('webhook/stripe', 'No session.customer — stripeCustomerId not updated');
   302→          }
   303→
   304→          await tx.user.update({
   305→            where: { id: userId },
   306→            data: updateData,
   307→          });
   308→
   309→          serverLog.info('webhook/stripe', 'User updated — plan: PREMIUM');
   310→
   311→          // Create subscription record (within transaction)
   312→          if (subscriptionId) {
   313→            await tx.subscription.upsert({
   314→              where: { stripeSubscriptionId: subscriptionId },
   315→              create: {
   316→                userId,
   317→                stripeSubscriptionId: subscriptionId,
   318→                stripePriceId,
   319→                status: 'active',
   320→                currentPeriodStart: periodStart,
   321→                currentPeriodEnd: periodEnd,
   322→              },
   323→              update: {
   324→                userId,
   325→                stripePriceId,
   326→                status: 'active',
   327→                currentPeriodStart: periodStart,
   328→                currentPeriodEnd: periodEnd,
   329→                cancelAtPeriodEnd: false,
   330→              },
   331→            });
   332→          } else {
   333→            // One-time payment or missing subscription ID — create a synthetic record
   334→            await tx.subscription.create({
   335→              data: {
   336→                userId,
   337→                stripeSubscriptionId: `checkout_${session.id}`,
   338→                stripePriceId,
   339→                status: 'active',
   340→                currentPeriodStart: periodStart,
   341→                currentPeriodEnd: periodEnd,
   342→              },
   343→            });
   344→          }
   345→        });
   346→
   347→        // Track premium upgrade completion (fire-and-forget, outside transaction)
   348→        trackEvent({ event: 'premium_upgrade_completed', userId }).catch(() => {});
   349→
   350→        // Send confirmation email (fire-and-forget, outside transaction)
   351→        try {
   352→          const user = await db.user.findUnique({ where: { id: userId } });
   353→          if (user?.email) {
   354→            sendSubscriptionConfirmedEmail(user.email, user.name || 'Amigo', 'Élite')
   355→              .then(() => serverLog.info('webhook/stripe', 'Subscription confirmation email sent'))
   356→              .catch((err) => serverLog.error('webhook/stripe', 'Subscription confirmation email failed', err));
   357→          }
   358→        } catch {
   359→          // Email lookup failure must not affect webhook response
   360→        }
   361→
   362→        break;
   363→      }
   364→
   365→      case 'customer.subscription.updated': {
   366→        const subscription = event.data.object as Stripe.Subscription;
   367→        serverLog.info('webhook/stripe', 'customer.subscription.updated', {
   368→          subId: subscription.id,
   369→          status: subscription.status,
   370→          eventId,
   371→        });
   372→
   373→        // Resolve userId from customer
   374→        let userId: string | null = null;
   375→        try {
   376→          const customer = await stripe.customers.retrieve(subscription.customer as string);
   377→          if ('metadata' in customer && !('deleted' in customer)) {
   378→            userId = customer.metadata.userId || null;
   379→          }
   380→        } catch (e) {
   381→          serverLog.error('webhook/stripe', 'Could not retrieve customer for subscription update', e);
   382→        }
   383→
   384→        // Fallback: find user by subscription in our DB
   385→        if (!userId) {
   386→          try {
   387→            const existingSub = await db.subscription.findUnique({
   388→              where: { stripeSubscriptionId: subscription.id },
   389→              select: { userId: true },
   390→            });
   391→            if (existingSub) {
   392→              userId = existingSub.userId;
   393→              serverLog.info('webhook/stripe', 'Resolved userId from existing subscription record');
   394→            }
   395→          } catch {
   396→            // DB lookup failed
   397→          }
   398→        }
   399→
   400→        if (!userId) {
   401→          serverLog.warn('webhook/stripe', 'customer.subscription.updated — cannot resolve userId', {
   402→            subId: subscription.id,
   403→          });
   404→          await markEventProcessed(event.id, event.type);
   405→          await cleanupOldEvents();
   406→          break;
   407→        }
   408→
   409→        // Update subscription record
   410→        // Stripe Basil/Dahlia: current_period_start/end moved from Subscription
   411→        // to SubscriptionItem. Use items.data[0] with defensive validation.
   412→        try {
   413→          const firstItem = subscription.items.data[0];
   414→          const itemPeriodStart = firstItem?.current_period_start;
   415→          const itemPeriodEnd = firstItem?.current_period_end;
   416→
   417→          if (!firstItem || typeof itemPeriodStart !== 'number' || typeof itemPeriodEnd !== 'number') {
   418→            serverLog.error('webhook/stripe', 'Subscription update — item missing period dates, skipping period update', undefined, {
   419→              subId: subscription.id,
   420→              hasItems: !!firstItem,
   421→              itemStartType: typeof itemPeriodStart,
   422→              itemEndType: typeof itemPeriodEnd,
   423→            });
   424→          }
   425→
   426→          await db.subscription.upsert({
   427→            where: { stripeSubscriptionId: subscription.id },
   428→            create: {
   429→              userId,
   430→              stripeSubscriptionId: subscription.id,
   431→              stripePriceId: subscription.items.data[0]?.price.id || '',
   432→              status: subscription.status,
   433→              currentPeriodStart: typeof itemPeriodStart === 'number' ? new Date(itemPeriodStart * 1000) : new Date(),
   434→              currentPeriodEnd: typeof itemPeriodEnd === 'number' ? new Date(itemPeriodEnd * 1000) : new Date(Date.now() + 30 * 24 * 60 * 60 * 1000),
   435→              cancelAtPeriodEnd: subscription.cancel_at_period_end,
   436→            },
   437→            update: {
   438→              status: subscription.status,
   439→              cancelAtPeriodEnd: subscription.cancel_at_period_end,
   440→              // BUG-C3 FIX: Also update currentPeriodStart on renewal.
   441→              // Previously only currentPeriodEnd was updated, so the start
   442→              // date stayed at the original subscription creation date
   443→              // instead of reflecting the current billing period start.
   444→              ...(typeof itemPeriodStart === 'number' ? { currentPeriodStart: new Date(itemPeriodStart * 1000) } : {}),
   445→              ...(typeof itemPeriodEnd === 'number' ? { currentPeriodEnd: new Date(itemPeriodEnd * 1000) } : {}),
   446→            },
   447→          });
   448→        } catch (e) {
   449→          serverLog.error('webhook/stripe', 'Failed to update subscription record', e, {
   450→            subId: subscription.id,
   451→          });
   452→        }
   453→
   454→        // ─── Sync user.plan with subscription status ───────────────────
   455→        // BUG-C5 FIX: Removed dead variable keepPremiumStatuses — it was
   456→        // declared but never referenced. The if-else branches below handle
   457→        // each status explicitly.
   458→        const downgradeStatuses = ['canceled', 'unpaid', 'incomplete_expired'];
   459→
   460→        if (downgradeStatuses.includes(subscription.status)) {
   461→          // Check if user has any other active/trialing subscription before downgrading
   462→          const otherActive = await db.subscription.findFirst({
   463→            where: {
   464→              userId,
   465→              status: { in: ['active', 'trialing'] },
   466→              stripeSubscriptionId: { not: subscription.id },
   467→            },
   468→          });
   469→          if (!otherActive) {
   470→            await db.user.update({
   471→              where: { id: userId },
   472→              data: { plan: 'FREE' },
   473→            });
   474→            serverLog.info('webhook/stripe', 'Subscription terminal-inactive — user downgraded to FREE', {
   475→              status: subscription.status,
   476→            });
   477→          } else {
   478→            serverLog.info('webhook/stripe', 'Subscription inactive but another active sub exists — keeping PREMIUM');
   479→          }
   480→        } else if (subscription.status === 'active') {
   481→          // Ensure user is marked PREMIUM if subscription is active
   482→          const user = await db.user.findUnique({ where: { id: userId } });
   483→          if (user && user.plan !== 'PREMIUM') {
   484→            await db.user.update({
   485→              where: { id: userId },
   486→              data: { plan: 'PREMIUM' },
   487→            });
   488→            serverLog.info('webhook/stripe', 'Subscription active — user restored to PREMIUM');
   489→          }
   490→        } else if (subscription.status === 'trialing') {
   491→          // BUG-B4 FIX: Trialing subscriptions must also promote FREE → PREMIUM.
   492→          // Previously, 'trialing' fell through with no action because
   493→          // keepPremiumStatuses was declared but never used. If the
   494→          // checkout.session.completed webhook was lost/delayed, the user
   495→          // remained FREE during their trial. Now we ensure PREMIUM here too.
   496→          const user = await db.user.findUnique({ where: { id: userId } });
   497→          if (user && user.plan !== 'PREMIUM') {
   498→            await db.user.update({
   499→              where: { id: userId },
   500→              data: { plan: 'PREMIUM' },
   501→            });
   502→            serverLog.info('webhook/stripe', 'Subscription trialing — user promoted to PREMIUM');
   503→          }
   504→        } else if (subscription.status === 'past_due') {
   505→          serverLog.info('webhook/stripe', 'Subscription past_due — keeping PREMIUM during Stripe retry period');
   506→        } else if (subscription.status === 'incomplete') {
   507→          serverLog.info('webhook/stripe', 'Subscription incomplete — keeping PREMIUM while payment processes');
   508→        }
   509→
   510→        break;
   511→      }
   512→
   513→      case 'customer.subscription.deleted': {
   514→        const subscription = event.data.object as Stripe.Subscription;
   515→        serverLog.info('webhook/stripe', 'customer.subscription.deleted', {
   516→          subId: subscription.id,
   517→          eventId,
   518→        });
   519→
   520→        // Resolve userId — try customer metadata first, then DB fallback
   521→        let userId: string | null = null;
   522→        try {
   523→          const customer = await stripe.customers.retrieve(subscription.customer as string);
   524→          if ('metadata' in customer && !('deleted' in customer)) {
   525→            userId = customer.metadata.userId || null;
   526→          }
   527→        } catch (e) {
   528→          serverLog.error('webhook/stripe', 'Could not retrieve customer for subscription deletion', e);
   529→        }
   530→
   531→        // Fallback: find user by subscription in our DB
   532→        if (!userId) {
   533→          try {
   534→            const existingSub = await db.subscription.findUnique({
   535→              where: { stripeSubscriptionId: subscription.id },
   536→              select: { userId: true },
   537→            });
   538→            if (existingSub) {
   539→              userId = existingSub.userId;
   540→              serverLog.info('webhook/stripe', 'Resolved userId from existing subscription record (deleted)');
   541→            }
   542→          } catch {
   543→            // DB lookup failed
   544→          }
   545→        }
   546→
   547→        if (!userId) {
   548→          serverLog.warn('webhook/stripe', 'customer.subscription.deleted — cannot resolve userId', {
   549→            subId: subscription.id,
   550→          });
   551→          await markEventProcessed(event.id, event.type);
   552→          await cleanupOldEvents();
   553→          break;
   554→        }
   555→
   556→        // Check if user has any other active subscription before downgrading
   557→        const otherActive = await db.subscription.findFirst({
   558→          where: {
   559→            userId,
   560→            status: { in: ['active', 'trialing'] },
   561→            stripeSubscriptionId: { not: subscription.id },
   562→          },
   563→        });
   564→
   565→        if (!otherActive) {
   566→          await db.user.update({
   567→            where: { id: userId },
   568→            data: { plan: 'FREE' },
   569→          });
   570→          serverLog.info('webhook/stripe', 'Subscription canceled, user downgraded to FREE');
   571→        } else {
   572→          serverLog.info('webhook/stripe', 'Subscription canceled but another active sub exists — keeping PREMIUM');
   573→        }
   574→
   575→        // Mark this subscription as canceled
   576→        await db.subscription.updateMany({
   577→          where: { stripeSubscriptionId: subscription.id },
   578→          data: { status: 'canceled' },
   579→        });
   580→
   581→        break;
   582→      }
   583→
   584→      case 'invoice.payment_failed': {
   585→        const invoice = event.data.object as Stripe.Invoice;
   586→        const subscriptionId = invoice.subscription as string | null;
   587→
   588→        serverLog.warn('webhook/stripe', 'invoice.payment_failed', {
   589→          invoiceId: invoice.id,
   590→          subscriptionId: subscriptionId || 'null',
   591→          attemptCount: invoice.attempt_count,
   592→          eventId,
   593→        });
   594→
   595→        // We don't downgrade on payment failure because Stripe will send
   596→        // customer.subscription.updated with past_due status, which we
   597→        // handle with a grace period. If all retries fail, Stripe
   598→        // transitions to canceled/unpaid, which triggers downgrade.
   599→        break;
   600→      }
   601→
   602→      default:
   603→        serverLog.info('webhook/stripe', 'Unhandled event type', { eventType: event.type, eventId: event.id });
   604→    }
   605→  } catch (error) {
   606→    const durationMs = Date.now() - start;
   607→    const errorDetails = serializeWebhookError(error);
   608→    serverLog.error('webhook/stripe', 'Webhook handler error', error, {
   609→      eventType,
   610→      eventId,
   611→      durationMs,
   612→      ...(checkoutSessionContext && { checkout: checkoutSessionContext }),
   613→      ...errorDetails,
   614→    });
   615→    // DON'T mark as processed — Stripe will retry, which is correct
   616→    // because the handler failed. If the error is persistent, we need
   617→    // to know about it (and Stripe will alert us after multiple failures).
   618→    return NextResponse.json({ error: 'Webhook handler failed' }, { status: 500 });
   619→  }
   620→
   621→  // ─── Mark event as processed + cleanup old events ────────────────
   622→  // Only mark as processed if the handler succeeded (no throw).
   623→  // This way, Stripe retries are useful for actual failures.
   624→  await markEventProcessed(event.id, event.type);
   625→  await cleanupOldEvents();
   626→
   627→  const durationMs = Date.now() - start;
   628→  serverLog.info('webhook/stripe', 'Webhook processed successfully', {
   629→    eventType,
   630→    eventId,
   631→    durationMs,
   632→  });
   633→
   634→  return NextResponse.json({ received: true });
   635→}
