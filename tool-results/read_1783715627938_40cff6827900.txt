   686→   735→function daysAgo(date: Date): number {
   687→   736→  const todayKey = getTodayDateKey();

   690→   739→  const dateMs = new Date(dateKey + 'T00:00:00').getTime();
   691→   740→  return Math.round((todayMs - dateMs) / 86400000);
   692→   741→}

   699→   748→/**
   700→   749→ * Extract a fragment from journal content for the mentor context.
   701→   750→ * Returns null if the content doesn't add information beyond the title.
   702→   751→ *
   703→   752→ * Rules:
   704→   753→ * - Max 180 characters
   705→   754→ * - Never cut words
   706→   755→ * - Remove line breaks and repeated spaces
   707→   756→ * - End with "…" when truncated
   708→   757→ * - No summarization, no reinterpretation — exact user text only
   709→   758→ * - Skip if content only repeats what the title already says
   710→   759→ */
   711→   760→function extractJournalFragment(title: string, content: string): string | null {
   712→   761→  // Clean content: remove line breaks, collapse repeated spaces, trim
   713→   762→  const cleaned = content
   714→   763→    .replace(/\n/g, ' ')
   715→   764→    .replace(/\s+/g, ' ')
   716→   765→    .trim();
   717→   766→
   718→   767→  if (!cleaned) return null;
   719→   768→
   720→   769→  // Normalize title for comparison (lowercase, no punctuation, collapsed spaces)
   721→   770→  const normalizeForComparison = (s: string) =>
   722→   771→    s.toLowerCase().replace(/[^\wáéíóúñü]/g, ' ').replace(/\s+/g, ' ').trim();
   723→   772→
   724→   773→  const normalizedTitle = normalizeForComparison(title);
   725→   774→  const normalizedCleaned = normalizeForComparison(cleaned);
   726→   775→
   727→   776→  // If content starts with the title (user often repeats title as first line),
   728→   777→  // strip the title prefix from content for comparison
   729→   778→  let contentForComparison = normalizedCleaned;
   730→   779→  if (normalizedCleaned.startsWith(normalizedTitle)) {
   731→   780→    contentForComparison = normalizedCleaned.slice(normalizedTitle.length).trim();
   732→   781→  }
   733→   782→
   734→   783→  // If what remains is empty or too short, content doesn't add information
   735→   784→  if (contentForComparison.length < 10) return null;
   736→   785→
   737→   786→  // Truncate to 180 chars without cutting words
   738→   787→  const MAX_LEN = 180;
   739→   788→  if (cleaned.length <= MAX_LEN) return cleaned;
   740→   789→
   741→   790→  // Find the last space before the limit
   742→   791→  const truncated = cleaned.slice(0, MAX_LEN);
   743→   792→  const lastSpace = truncated.lastIndexOf(' ');
   744→   793→  if (lastSpace <= 0) return cleaned.slice(0, MAX_LEN) + '…';
   745→   794→
   746→   795→  return truncated.slice(0, lastSpace) + '…';
   747→   796→}
