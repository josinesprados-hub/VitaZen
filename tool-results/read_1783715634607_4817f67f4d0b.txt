    90→function getScoreColor(score: number): string {
    91→  if (score >= 75) return '#22c55e';
    92→  if (score >= 50) return '#b8995e';   // muted champagne — less orange, more editorial
    93→  if (score >= 30) return '#c49856';   // warm amber — not neon, still warm
    94→  return '#ef4444';
    95→}
    96→
    97→function getScoreLabel(score: number): string {
    98→  if (score >= 80) return 'Muy activo';
    99→  if (score >= 60) return 'Actividad moderada';
   100→  if (score >= 40) return 'Semana tranquila';
   101→  return 'Poca actividad';
   102→}
   103→
   104→function getCategoryHref(category: string): string {
   105→  const map: Record<string, string> = {
   106→    'hábitos': '/imperio/disciplina',
   107→    'meditación': '/imperio/mente',
   108→    'emociones': '/checkin',
   109→    'energía': '/imperio/energia',
   110→    'estrés': '/imperio/mente',
   111→    'diario': '/imperio/crecimiento',
   112→    'finanzas': '/imperio/riqueza',
   113→    'nutrición': '/imperio/energia',
   114→    'bienestar': '/insights',
   115→    'consistencia': '/insights',
   116→    'actividad': '/insights',
   117→    'imperios': '/insights',
   118→    'recomendación': '/imperio/mentor',
   119→  };
   120→  return map[category.toLowerCase()] || '/insights';
   121→}
   122→
   123→function getInsightBorderClass(type: string): string {
   124→  switch (type) {
   125→    case 'positive': return 'border-[#22c55e]/20 hover:border-[#22c55e]/40';
   126→    case 'warning': return 'border-champagne-warm/20 hover:border-champagne-warm/40';
   127→    case 'trend': return 'border-champagne-soft/20 hover:border-champagne-soft/40';
   128→    default: return 'border-[#1a1a1a] hover:border-[#2a2a2a]';
   129→  }
   130→}
   131→
   132→function getInsightBadgeClass(type: string): string {
   133→  switch (type) {
   134→    case 'positive': return 'bg-[#22c55e]/10 text-[#22c55e]';
   135→    case 'warning': return 'bg-champagne-warm/10 text-champagne-warm';
   136→    case 'trend': return 'bg-champagne-soft/10 text-champagne-soft';
   137→    default: return 'bg-[#1a1a1a] text-[#666]';
   138→  }
   139→}
   140→
   141→function TrendIndicator({ value, label }: { value: number; label: string }) {
   142→  if (value > 0.2) {
   143→    return (
   144→      <span className="flex items-center gap-1 text-[11px] text-[#22c55e]">
   145→        <TrendingUp size={12} /> {label} ↑
   146→      </span>
   147→    );
   148→  }
   149→  if (value < -0.2) {
   150→    return (
   151→      <span className="flex items-center gap-1 text-[11px] text-champagne-warm">
   152→        <TrendingDown size={12} /> {label} ↓
   153→      </span>
   154→    );
   155→  }
   156→  return (
   157→    <span className="flex items-center gap-1 text-[11px] text-[#666]">
   158→      <Minus size={12} /> {label} —
   159→    </span>
   160→  );
   161→}
