   131→   131→  const thirtyDaysAgo = new Date(now.getTime() - 30 * 86400000);
   132→   132→  const ninetyDaysAgo = new Date(now.getTime() - 90 * 86400000);
   133→   133→
   134→   134→  // Previous week window for trend comparison
   135→   135→  const fourteenDaysAgo = new Date(now.getTime() - 14 * 86400000);
   136→   136→
   137→   137→  // Run all queries in parallel for performance
   138→   138→  const [
   139→   139→    recentCheckins,
   140→   140→    habitStreaks,
   141→   141→    recentMeditations,
   142→   142→    recentJournals,

   184→   184→    // Only PREMIUM loads 'content' for intelligent fragment extraction.

   187→   187→      orderBy: { createdAt: 'desc' },
   188→   188→      take: isPremium ? 3 : 1,
   189→   189→      select: isPremium
   190→   190→        ? { title: true, content: true, mood: true, createdAt: true }

   193→   193→
   194→   194→    // Recent conversation threads: FREE gets 1, PREMIUM gets 3
   195→   195→    // L-7 FIX: Removed 'include: { messages: ... }' — lastMessage was fetched
   196→   196→    // but never used in any context formatter. The join was wasted DB work

   202→   202→      select: { id: true, title: true, updatedAt: true, archived: true },
   203→   203→    }),

   205→   205→    // Empire progress: PREMIUM only
   206→   206→    isPremium
   207→   207→      ? db.empireProgress.findMany({ where: { userId } })

   210→   210→    // Weekly meditation dates: PREMIUM only (for distinct-day counting)

   213→   213→      : Promise.resolve([] as { completedAt: Date }[]),

   218→   218→      : Promise.resolve([] as { createdAt: Date }[]),
   219→   219→
   220→   220→    // Weekly checkins: both tiers (for consistency + distinct-day counting + emotional state)
   221→   221→    db.dailyCheckin.findMany({ where: { userId, date: { gte: sevenDaysAgo } }, select: { date: true, emotion: true, energy: true, focus: true, stress: true } }),
   222→   222→
   223→   223→    // Weekly habit log dates: last 7 days (for distinct-day counting)
   224→   224→    db.habitLog.findMany({
   225→   225→      where: { userId, lastCompletedAt: { gte: sevenDaysAgo } },
   226→   226→      select: { lastCompletedAt: true },
   227→   227→    }),
   228→   228→
   229→   229→    // Previous week checkins (for trend + emotional state): both tiers
   230→   230→    db.dailyCheckin.findMany({
   231→   231→      where: { userId, date: { gte: fourteenDaysAgo, lt: sevenDaysAgo } },
   232→   232→      select: { date: true, emotion: true, energy: true, focus: true, stress: true },
   233→   233→    }),
   234→   234→
   235→   235→    // User data
   236→   236→    db.user.findUnique({
   237→   237→      where: { id: userId },
   238→   238→      select: { name: true, plan: true },
   239→   239→    }),
   240→   240→
   241→   241→    // Onboarding data: PREMIUM only
   242→   242→    isPremium
   243→   243→      ? db.onboardingData.findUnique({
   244→   244→          where: { userId },
   245→   245→          select: {
   246→   246→            goals: true,
   247→   247→            primaryFocus: true,
   248→   248→            stressLevel: true,
   249→   249→            energyLevel: true,
   250→   250→            focusLevel: true,
   251→   251→            initialHabits: true,
   252→   252→          },
   253→   253→        })
   254→   254→      : Promise.resolve(null),
   255→   255→
   256→   256→    // Wellness logs: PREMIUM only — sleep quality + notes (dedup: mood/energy/stress already in DailyCheckin)
   257→   257→    // No take limit: full 14-day data needed for emotional state; display is sliced to 7 later
   258→   258→    isPremium
   259→   259→      ? db.wellnessLog.findMany({
   260→   260→          where: { userId, date: { gte: fourteenDaysAgo } },
   261→   261→          orderBy: { date: 'desc' },
   262→   262→          select: { date: true, sleep: true, mood: true, stress: true, notes: true },
   263→   263→        })
   264→   264→      : Promise.resolve([]),
   265→   265→
   266→   266→    // Finance logs: PREMIUM only — spending mood patterns + contexto
   267→   267→    isPremium
   268→   268→      ? db.financeLog.findMany({
   269→   269→          where: { userId, date: { gte: thirtyDaysAgo }, type: 'expense' },
   270→   270→          orderBy: { date: 'desc' },
   271→   271→          take: 15,
   272→   272→          select: { type: true, category: true, mood: true, contexto: true, date: true },
   273→   273→        })
   274→   274→      : Promise.resolve([]),
   275→   275→
   276→   276→    // Weekly wellness log dates: PREMIUM only (for distinct-day counting)
   277→   277→    isPremium
   278→   278→      ? db.wellnessLog.findMany({ where: { userId, date: { gte: sevenDaysAgo } }, select: { date: true } })
   279→   279→      : Promise.resolve([] as { date: Date }[]),
   280→   280→
   281→   281→    // Weekly nutrition log dates: PREMIUM only (for distinct-day counting)
   282→   282→    isPremium
   283→   283→      ? db.nutritionLog.findMany({ where: { userId, date: { gte: sevenDaysAgo } }, select: { date: true } })
   284→   284→      : Promise.resolve([] as { date: Date }[]),
   285→   285→
   286→   286→    // Previous week habit logs: PREMIUM only (for emotional state prevWeekHabits)
   287→   287→    isPremium
   288→   288→      ? db.habitLog.findMany({ where: { userId, lastCompletedAt: { gte: fourteenDaysAgo, lt: sevenDaysAgo } }, select: { lastCompletedAt: true } })
   289→   289→      : Promise.resolve([] as { lastCompletedAt: Date | null }[]),
   290→   290→
   291→   291→    // Previous week meditations: PREMIUM only (for emotional state prevWeekMeditations)
   292→   292→    isPremium
   293→   293→      ? db.meditationSession.findMany({ where: { userId, completedAt: { gte: fourteenDaysAgo, lt: sevenDaysAgo } }, select: { completedAt: true } })
   294→   294→      : Promise.resolve([] as { completedAt: Date }[]),
   295→   295→
   296→   296→    // Previous week journals: PREMIUM only (for emotional state prevWeekJournals)
   297→   297→    isPremium
   298→   298→      ? db.journalEntry.findMany({ where: { userId, createdAt: { gte: fourteenDaysAgo, lt: sevenDaysAgo } }, select: { createdAt: true } })
   299→   299→      : Promise.resolve([] as { createdAt: Date }[]),
   300→   300→
   301→   301→    // Previous week nutrition logs: PREMIUM only (for emotional state prevWeekNutrition)
   302→   302→    isPremium
   303→   303→      ? db.nutritionLog.findMany({ where: { userId, date: { gte: fourteenDaysAgo, lt: sevenDaysAgo } }, select: { date: true } })
   304→   304→      : Promise.resolve([] as { date: Date }[]),
   305→   305→  ]);
   306→   306→
   307→   307→  // Derive counts from date arrays (for weeklyActivity display)
   308→   308→  const weeklyMeditationCount = weeklyMeditations.length;
   309→   309→  const weeklyJournalCount = weeklyJournals.length;
   310→   310→  const weeklyCheckinCount = weeklyCheckins.length;
   311→   311→  const weeklyHabitLogCount = weeklyHabitLogs.length;
   312→   312→  const weeklyWellnessCount = weeklyWellnessLogs.length;
   313→   313→  const weeklyNutritionCount = weeklyNutritionLogs.length;
   314→   314→
   315→   315→  // Weekly habit completions (PREMIUM only for display, but count for consistency)
   316→   316→  const weeklyHabits = isPremium
   317→   317→    ? habitStreaks.filter(h => {
   318→   318→        if (!h.lastCompletedAt) return false;
   319→   319→        return h.lastCompletedAt >= sevenDaysAgo;
   320→   320→      }).length
   321→   321→    : weeklyHabitLogCount;
   322→   322→
   323→   323→  // Compute consistency signal: count DISTINCT days with any activity
   324→   324→  const activeDaySet = new Set<string>();
   325→   325→  for (const c of weeklyCheckins) activeDaySet.add(toDateKey(c.date));
   326→   326→  for (const h of weeklyHabitLogs) {
   327→   327→    if (h.lastCompletedAt) activeDaySet.add(toDateKey(h.lastCompletedAt));
   328→   328→  }
   329→   329→  for (const m of weeklyMeditations) activeDaySet.add(toDateKey(m.completedAt));
   330→   330→  for (const j of weeklyJournals) activeDaySet.add(toDateKey(j.createdAt));
   331→   331→  for (const w of weeklyWellnessLogs) activeDaySet.add(toDateKey(w.date));
   332→   332→  for (const n of weeklyNutritionLogs) activeDaySet.add(toDateKey(n.date));
   333→   333→  const activeDaysThisWeek = Math.min(activeDaySet.size, 7);
   334→   334→
   335→   335→  const prevWeekCheckinCount = prevWeekCheckins.length;
   336→   336→
   337→   337→  // M-7: Compute trend based on aggregate activity, not just check-ins.
   338→   338→  // PREMIUM has full prev-week data → compare activeDaysThisWeek vs activeDaysPrevWeek.
   339→   339→  // FREE only has prev-week checkins → keep checkin-based trend (no other prev-week data available).
   340→   340→  let trend: 'improving' | 'stable' | 'declining' | 'starting' = 'stable';
   341→   341→  if (isPremium) {
   342→   342→    const prevActiveDaySet = new Set<string>();
   343→   343→    for (const c of prevWeekCheckins) prevActiveDaySet.add(toDateKey(c.date));
   344→   344→    for (const h of prevWeekHabitLogs) {
   345→   345→      if (h.lastCompletedAt) prevActiveDaySet.add(toDateKey(h.lastCompletedAt));
   346→   346→    }
   347→   347→    for (const m of prevWeekMeditations) prevActiveDaySet.add(toDateKey(m.completedAt));

   351→   351→    for (const n of prevWeekNutritionLogs) prevActiveDaySet.add(toDateKey(n.date));
   352→   352→    const activeDaysPrevWeek = Math.min(prevActiveDaySet.size, 7);
   353→   353→
   354→   354→    if (activeDaysThisWeek <= 1 && activeDaysPrevWeek <= 1) {
   355→   355→      trend = 'starting';
   356→   356→    } else if (activeDaysThisWeek > activeDaysPrevWeek + 1) {
   357→   357→      trend = 'improving';
   358→   358→    } else if (activeDaysThisWeek < activeDaysPrevWeek - 1) {
   359→   359→      trend = 'declining';
   360→   360→    }
   361→   361→  } else {
   362→   362→    if (weeklyCheckinCount <= 1 && prevWeekCheckinCount <= 1) {
   363→   363→      trend = 'starting';
   364→   364→    } else if (weeklyCheckinCount > prevWeekCheckinCount + 1) {
   365→   365→      trend = 'improving';
   366→   366→    } else if (weeklyCheckinCount < prevWeekCheckinCount - 1) {
   367→   367→      trend = 'declining';
   368→   368→    }
   369→   369→  }
   370→   370→
   371→   371→  // Parse onboarding data safely (PREMIUM only; null for FREE)
   372→   372→  let onboardingData: UserContext['onboardingData'] = null;
   373→   373→  if (onboardingRow) {
   374→   374→    let parsedGoals: string[] = [];
   375→   375→    try { parsedGoals = JSON.parse(onboardingRow.goals); } catch { /* keep empty */ }
   376→   376→    let parsedHabits: string[] = [];
   377→   377→    try { parsedHabits = JSON.parse(onboardingRow.initialHabits); } catch { /* keep empty */ }

   379→   379→      goals: Array.isArray(parsedGoals) ? parsedGoals : [],
   380→   380→      primaryFocus: onboardingRow.primaryFocus || null,
   381→   381→      stressLevel: onboardingRow.stressLevel,
   382→   382→      energyLevel: onboardingRow.energyLevel,
   383→   383→      focusLevel: onboardingRow.focusLevel,
   384→   384→      initialHabits: Array.isArray(parsedHabits) ? parsedHabits : [],
   385→   385→    };
   386→   386→  }
   387→   387→
   388→   388→  // Fetch emotional state from the official engine (PREMIUM only; null for FREE)
   389→   389→  // This is the single source of truth — no recalculation, no duplication.
   390→   390→  // M-6: Reuse data already fetched in this Promise.all to avoid redundant gatherData() call.
   391→   391→  let emotionalState: UserContext['emotionalState'] = null;
   392→   392→  if (isPremium) {
   393→   393→    try {
   394→   394→      // Build RawData from already-fetched queries — no extra DB access
   395→   395→      const rawDataForEmotionalState: RawData = {
   396→   396→        thisWeekCheckins: weeklyCheckins,
   397→   397→        prevWeekCheckins,
   398→   398→        thisWeekHabits: weeklyHabitLogs,
   399→   399→        prevWeekHabits: prevWeekHabitLogs,
   400→   400→        allHabits: [],
   401→   401→        thisWeekMeditations: weeklyMeditations,
   402→   402→        prevWeekMeditations,
   403→   403→        thisWeekJournals: weeklyJournals,
   404→   404→        prevWeekJournals,
   405→   405→        thisWeekWellness: wellnessLogRows.filter(w => w.date >= sevenDaysAgo),
   406→   406→        prevWeekWellness: wellnessLogRows.filter(w => w.date < sevenDaysAgo),
   407→   407→        thisWeekNutrition: weeklyNutritionLogs,
   408→   408→        prevWeekNutrition: prevWeekNutritionLogs,
   409→   409→        thisWeekFinance: [],
   410→   410→        prevWeekFinance: [],
   411→   411→        empireProgress: [],
   412→   412→      };
   413→   413→
   414→   414→      const es = await getEmotionalState(userId, plan, rawDataForEmotionalState);
   415→   415→      emotionalState = {
   416→   416→        status: es.status,
   417→   417→        statusLabel: es.statusLabel,
   418→   418→        statusDescription: es.statusDescription,
   419→   419→        summary: es.summary,
   420→   420→        recommendation: es.recommendation,
   421→   421→      };
   422→   422→    } catch {
   423→   423→      // Non-blocking: if emotional state fails, continue without it
   424→   424→    }
   425→   425→  }
   426→   426→
   427→   427→  // Monthly closure records: consume existing closure records (PREMIUM only; empty for FREE)
   428→   428→  // No recalculation, no summaries — just the fact that closures exist.
   429→   429→  const recentClosures = isPremium
   430→   430→    ? await db.monthlyClosure.findMany({
   431→   431→        where: { userId },
   432→   432→        select: {
   433→   433→          month: true,
   434→   434→          reflection: true,
   435→   435→          reflectedAt: true,
   436→   436→          summaryViewedAt: true,
   437→   437→        },
   438→   438→        orderBy: { month: 'desc' },
   439→   439→        take: 3,
   440→   440→      })
   441→   441→    : [];
   442→   442→
   443→   443→  // Life stage detection: consume the official engine result (PREMIUM only; null for FREE)
   444→   444→  // No recalculation, no inference — the engine is the single source of truth for life stages.
   445→   445→  let lifeStage: UserContext['lifeStage'] = null;

   752→   752→ * Rules:
   753→   753→ * - Max 180 characters

   760→   760→function extractJournalFragment(title: string, content: string): string | null {
