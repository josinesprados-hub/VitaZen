   315→   315→// ─── Detección de repetición en historial ───
   316→   316→
   317→   317→function detectRepetition(message: string, history: { role: string; content: string }[]): boolean {
   318→   318→  const lower = message.toLowerCase();
   319→   319→  const words = new Set(lower.split(/\s+/).filter(w => w.length >= 4));
   320→   320→  if (words.size === 0) return false;
   321→   321→
   322→   322→  // Check last 3 user messages for high overlap
   323→   323→  const recentUser = history
   324→   324→    .filter(m => m.role === 'user')
   325→   325→    .slice(-3);
   326→   326→
   327→   327→  for (const msg of recentUser) {
   328→   328→    const msgLower = msg.content.toLowerCase();
   329→   329→    const msgWords = new Set(msgLower.split(/\s+/).filter(w => w.length >= 4));
   330→   330→    if (msgWords.size === 0) continue;
   331→   331→
   332→   332→    let overlap = 0;
   333→   333→    for (const w of words) {
   334→   334→      if (msgWords.has(w)) overlap++;
   335→   335→    }
   336→   336→
   337→   337→    const similarity = overlap / Math.min(words.size, msgWords.size);
   338→   338→    if (similarity > 0.7) return true;
   339→   339→  }
   340→   340→
   341→   341→  return false;
   342→   342→}

   344→   344→// ─── Detección de dominios en un texto ───
   345→   345→
   346→   346→function detectDomains(text: string): Domain[] {
   347→   347→  const lower = text.toLowerCase();
   348→   348→  const detected: { domain: Domain; score: number }[] = [];
   349→   349→
   350→   350→  for (const [domain, keywords] of Object.entries(DOMAIN_KEYWORDS)) {
   351→   351→    let score = 0;
   352→   352→    for (const kw of keywords) {
   353→   353→      if (lower.includes(kw)) score++;
   354→   354→    }
   355→   355→    if (score > 0) {
   356→   356→      detected.push({ domain: domain as Domain, score });
   357→   357→    }
   358→   358→  }
   359→   359→
   360→   360→  detected.sort((a, b) => b.score - a.score);
   361→   361→  return detected.map(d => d.domain);
   362→   362→}

   395→   395→  }
   396→   396→  if (/(?:estrés|ansiedad|miedo|frustrado|triste|agotado|abrumado|cansado|nervioso)/i.test(messageLower)) {
   397→   397→    scores.set('emotional_state', 1.0);
   398→   398→  }
   399→   399→
   400→   400→  // Life stage: relevante cuando hay contexto temporal amplio o cambio
   401→   401→  scores.set('life_stage', 0.1);
   402→   402→  if (input.lifeStage) {
   403→   403→    if (['exhaustion', 'intensity', 'dispersion'].includes(input.lifeStage.flavor)) {
   404→   404→      scores.set('life_stage', 0.6);
   405→   405→    }
   406→   406→  }
   407→   407→  if (/(?:últimamente|estas\s+(?:semanas|últimas)|mes\s+pasado|tiempo|hace\s+(?:un\s+rato|tiempo))/i.test(messageLower)) {
   408→   408→    scores.set('life_stage', 0.7);
   409→   409→  }
   410→   410→
   411→   411→  // Silent memories: baja relevancia por defecto, más si habla de tiempo/presencia
   412→   412→  scores.set('silent_memories', 0.1);
   413→   413→  if (input.silentMemories.length > 0 && /(?:regreso|volví|hace\s+(?:tiempo|mucho)|otra\s+vez|otra\s+vez)/i.test(messageLower)) {
   414→   414→    scores.set('silent_memories', 0.7);
   415→   415→  }
   416→   416→
   417→   417→  // Goals: muy relevante cuando habla de objetivos/hábitos/compromisos
   418→   418→  scores.set('goals', 0.15);
   419→   419→  if (input.hasGoals && /(?:objetivo|meta|compromiso|quiero|voy\s+a|estoy\s+intentando|mi\s+propósito|racha|constancia|hábito)/i.test(messageLower)) {
   420→   420→    scores.set('goals', 0.95);
   421→   421→  }
   422→   422→  if (domains.includes('disciplina')) scores.set('goals', Math.max(scores.get('goals') ?? 0, 0.7));
   423→   423→
   424→   424→  // Continuity: relevante cuando retoma temas o hay conexión histórica
   425→   425→  scores.set('continuity', 0.1);
   426→   426→  if (input.hasContinuity) scores.set('continuity', 0.6);
   427→   427→  if (/(?:lo\s+(?:que\s+)?(?:dijiste|hablamos)|el\s+otro\s+día|como\s+(?:te\s+)?dije|estuve\s+pensando)/i.test(messageLower)) {
   428→   428→    scores.set('continuity', 0.9);
   429→   429→  }
   430→   430→
   431→   431→  // Patterns: relevante cuando habla de conexiones entre áreas
   432→   432→  scores.set('patterns', 0.05);
   433→   433→  if (input.hasPatterns && domains.length >= 2) {
   434→   434→    scores.set('patterns', 0.6);
   435→   435→  }
   436→   436→
   437→   437→  // Monthly closure: baja relevancia, solo si habla del mes/cierre
   438→   438→  scores.set('monthly_closure', 0.05);
   439→   439→  if (input.hasMonthlyClosure && /(?:mes\s+pasado|este\s+mes|cierre|resumen\s+mensual|balance)/i.test(messageLower)) {
   440→   440→    scores.set('monthly_closure', 0.8);
   441→   441→  }
   442→   442→
   443→   443→  // Domain-specific systems
   444→   444→  scores.set('habits', domains.includes('disciplina') ? 0.8 : 0.1);
   445→   445→  scores.set('meditation', domains.includes('mente') ? 0.7 : 0.1);
   446→   446→  scores.set('nutrition', domains.includes('energia') ? 0.5 : 0.05);
   447→   447→  scores.set('finances', domains.includes('riqueza') ? 0.8 : 0.05);
   448→   448→
   449→   449→  // FREE: reducir número de sistemas relevantes
   450→   450→  const threshold = isPremium ? 0.4 : 0.7;
   451→   451→  const maxRelevant = isPremium ? 7 : 3;
   452→   452→
   453→   453→  const relevant = allSystems
   454→   454→    .filter(s => (scores.get(s) ?? 0) >= threshold)
   455→   455→    .sort((a, b) => (scores.get(b) ?? 0) - (scores.get(a) ?? 0))
   456→   456→    .slice(0, maxRelevant);
   457→   457→
   458→   458→  const ignored = allSystems.filter(s => !relevant.includes(s));
   459→   459→
   460→   460→  return { relevant, ignored };
   461→   461→}

   465→   465→function computeDepth(input: ReasoningInput, domains: Domain[], isHybrid: boolean): ReasoningDepth {
   466→   466→  const isPremium = input.plan === 'PREMIUM';
   467→   467→  const msgLen = input.message.length;
   468→   468→
   469→   469→  let depthScore = 0;
   470→   470→
   471→   471→  // Mensajes largos y de múltiples dominios → más profundidad
   472→   472→  if (msgLen > 200) depthScore += 1;
   473→   473→  if (domains.length >= 2) depthScore += 1;
   474→   474→  if (isHybrid) depthScore += 1;
   475→   475→
   476→   476→  // Estado emocional negativo → más profundidad necesaria
   477→   477→  if (input.emotionalState?.status === 'sobrecargado') depthScore += 1;
   478→   478→  if (input.emotionalState?.status === 'en_progreso') depthScore += 0.5;
   479→   479→
   480→   480→  // FREE: limitar profundidad
   481→   481→  if (!isPremium) {
   482→   482→    return depthScore >= 1.5 ? 'moderado' : 'superficial';
   483→   483→  }
   484→   484→
   485→   485→  // PREMIUM: toda la profundidad disponible
   486→   486→  if (depthScore >= 2.5) return 'profundo';
   487→   487→  if (depthScore >= 1) return 'moderado';
   488→   488→  return 'superficial';
   489→   489→}

   493→   493→function computeLength(need: PrimaryNeed, depth: ReasoningDepth, isPremium: boolean, isRepetition: boolean): RecommendedLength {
   494→   494→  // Repetición: más corto siempre
   495→   495→  if (isRepetition) return 'corta';
   496→   496→
   497→   497→  // FREE: priorizar respuestas cortas
   498→   498→  if (!isPremium) {
   499→   499→    if (need === 'descarga_emocional') return 'media';
   500→   500→    if (need === 'celebracion') return 'corta';
   501→   501→    return 'corta';
   502→   502→  }
   503→   503→
   504→   504→  // PREMIUM
   505→   505→  if (depth === 'profundo') return need === 'descarga_emocional' ? 'media' : 'larga';
   506→   506→  if (depth === 'moderado') return 'media';
   507→   507→  return 'corta';
   508→   508→}

   512→   512→function detectCelebration(message: string, history: { role: string; content: string }[]): boolean {
   513→   513→  // Verificar que el usuario no celebró lo mismo recientemente
   514→   514→  const isCelebrationMsg = /(?:he\s+(?:logrado|conseguido|completado|terminado|finalicé)|ya\s+(?:lo|lo\s+he)\s+(?:hice|logré|conseguí|terminé)|meta\s+(?:cumplida|alcanzada|lograda)|objetivo\s+(?:cumplido|alcanzado))/i.test(message);
   515→   515→
   516→   516→  if (!isCelebrationMsg) return false;
   517→   517→
   518→   518→  // No celebrar si ya se celebró el mismo tema en las últimas 3 interacciones
   519→   519→  const recentAssistant = history.filter(m => m.role === 'assistant').slice(-3);
   520→   520→  for (const msg of recentAssistant) {
   521→   521→    if (/(?:bien\s+hecho|enhorabuena|excelente|genial| Increíble|eso\s+es\s+gran|mereces|muy\s+bien|felicitaciones)/i.test(msg.content)) {
   522→   522→      return false;
   523→   523→    }
   524→   524→  }
   525→   525→
   526→   526→  return true;
   527→   527→}

   542→   542→

   546→   546→// ─── Calcular confianza ───
   547→   547→
   548→   548→function computeConfidence(
   549→   549→  hasPrimaryMatch: boolean,
   550→   550→  domainCount: number,
   551→   551→  isHybrid: boolean
   552→   552→): number {
   553→   553→  let confidence = 0.3; // base
   554→   554→
   555→   555→  if (hasPrimaryMatch) confidence += 0.35;
   556→   556→  if (domainCount > 0) confidence += 0.15;
   557→   557→  if (isHybrid) confidence += 0.1;
   558→   558→  if (domainCount >= 2) confidence += 0.1;
   559→   559→
   560→   560→  return Math.min(confidence, 1.0);
   561→   561→}

   565→   565→function determineAskQuestion(message: string, need: PrimaryNeed, isPremium: boolean): { ask: boolean; max: 0 | 1 } {
   566→   566→  // Nunca preguntar en estos casos
   567→   567→  if (need === 'celebracion') return { ask: false, max: 0 };
   568→   568→  if (need === 'urgencia') return { ask: false, max: 0 };
   569→   569→  if (need === 'descarga_emocional') return { ask: false, max: 0 };
   570→   570→
   571→   571→  // Preguntas directas: responder, no preguntar
   572→   572→  if (/\?$/.test(message.trim())) return { ask: false, max: 0 };
   573→   573→  if (/^(?:qué|cómo|cuál|cuándo|dónde|quién|cuánto|por\s+qué)/i.test(message.trim())) {
   574→   574→    return { ask: false, max: 0 };
   575→   575→  }
   576→   576→
   577→   577→  // FREE: minimizar preguntas (mensajes limitados)
   578→   578→  if (!isPremium) return { ask: false, max: 0 };
   579→   579→
   580→   580→  // Detectar si el usuario está pidiendo implícitamente una pregunta
   581→   581→  const shouldAsk = ASK_QUESTION_INDICATORS.some(p => p.test(message));
   582→   582→  return { ask: shouldAsk, max: 1 };
   583→   583→}

   609→   609→  if (decision.askQuestion) {

   615→   615→  // Sistemas relevantes para contexto

   619→   619→      life_stage: 'etapa de vida',

   628→   628→      finances: 'finanzas',

   630→   630→    const labels = decision.relevantSystems.map(s => systemLabels[s]).filter(Boolean);

   632→   632→  }
   633→   633→
   634→   634→  parts.push('── Fin de razonamiento ──');
   635→   635→  parts.push('');
   636→   636→
   637→   637→  return parts.join('\n');
   638→   638→}

   640→   640→// ─── Función principal ───
   641→   641→
   642→   642→/**
   643→   643→ * Reasoning Engine
   644→   644→ *
   645→   645→ * Analiza el mensaje del usuario y el contexto disponible para decidir
   646→   646→ * cómo debe responder el mentor. No genera texto. Solo devuelve una
   647→   647→ * estructura de decisión que se inyecta en el prompt del sistema.
   648→   648→ *
   649→   649→ * REGLAS:
   650→   650→ * 1. No hace consultas nuevas — usa solo datos del request actual
   651→   651→ * 2. No llama a Groq ni a ningún modelo externo
   652→   652→ * 3. Si falla, retorna null (non-blocking)
   653→   653→ * 4. Latencia < 1 ms (solo lógica de strings)
   654→   654→ * 5. FREE: razonamiento simplificado
   655→   655→ * 6. PREMIUM: toda la inteligencia disponible
   656→   656→ */
   657→   657→export function reason(input: ReasoningInput): ReasoningOutput | null {
   658→   658→  try {
   659→   659→    const message = input.message;
   660→   660→    const isPremium = input.plan === 'PREMIUM';
   661→   661→
   662→   662→    // ── Paso 1: Detectar si es un mensaje híbrido ──
   663→   663→    let isHybrid = false;
   664→   664→    let hybridOverride: { primary: PrimaryNeed; secondary: PrimaryNeed } | null = null;
   665→   665→
   666→   666→    for (const hp of HYBRID_PATTERNS) {
   667→   667→      if (hp.regex.test(message)) {
   668→   668→        isHybrid = true;
   669→   669→        hybridOverride = { primary: hp.primaryNeed, secondary: hp.secondaryNeed };
   670→   670→        break;
   671→   671→      }
   672→   672→    }
   673→   673→
   674→   674→    // ── Paso 2: Detectar necesidad principal por patrones ──
   675→   675→    let matchedPattern: IntentPattern | null = null;
   676→   676→    let primaryNeed: PrimaryNeed = 'orientacion'; // default
   677→   677→    let secondaryNeed: PrimaryNeed | null = null;
   678→   678→
   679→   679→    if (hybridOverride) {
   680→   680→      primaryNeed = hybridOverride.primary;
   681→   681→      secondaryNeed = hybridOverride.secondary;
   682→   682→    } else {
   683→   683→      for (const pattern of INTENT_PATTERNS) {
   684→   684→        if (pattern.regex.test(message)) {
   685→   685→          matchedPattern = pattern;
   686→   686→          primaryNeed = pattern.need;
   687→   687→          break;
   688→   688→        }
   689→   689→      }
   690→   690→    }
   691→   691→
   692→   692→    // ── Paso 3: Detectar dominios ──
   693→   693→    const domains = detectDomains(message);
   694→   694→    const priorityDomain = domains[0] || 'general';
   695→   695→    const secondaryDomain = domains[1] || null;
   696→   696→
   697→   697→    // ── Paso 4: Evaluar sistemas relevantes ──
   698→   698→    const { relevant, ignored } = evaluateSystems(input, domains);
   699→   699→
   700→   700→    // ── Paso 5: Calcular profundidad ──
   701→   701→    const depth = computeDepth(input, domains, isHybrid);
   702→   702→
   703→   703→    // ── Paso 6: Detectar repetición ──
   704→   704→    const isRepetition = detectRepetition(message, input.recentHistory);
   705→   705→
   706→   706→    // ── Paso 7: Determinar flags de comportamiento ──
   707→   707→    const shouldCelebrate = detectCelebration(message, input.recentHistory);
   708→   708→    const shouldChallenge = detectChallenge(message, input.emotionalState);
   709→   709→    const shouldReflect = detectReflection(message, primaryNeed);
   710→   710→    const shouldBePractical = primaryNeed === 'orientacion' || primaryNeed === 'urgencia';
   711→   711→
   712→   712→    // ── Paso 8: Determinar tono ──
   713→   713→    let tone: RecommendedTone = matchedPattern?.tone || 'directo';
   714→   714→
   715→   715→    // Ajustar tono según estado emocional
   716→   716→    if (input.emotionalState?.status === 'sobrecargado' && tone !== 'empatico') {
   717→   717→      tone = 'empatico';
   718→   718→    }
   719→   719→    if (isRepetition && tone === 'desafiante') {
   720→   720→      tone = 'directo';
   721→   721→    }
   722→   722→
   723→   723→    // ── Paso 9: Determinar longitud ──
   724→   724→    const length = computeLength(primaryNeed, depth, isPremium, isRepetition);
   725→   725→
   726→   726→    // ── Paso 10: Preguntas ──
   727→   727→    const { ask, max } = determineAskQuestion(message, primaryNeed, isPremium);
   728→   728→
   729→   729→    // ── Paso 11: Objetivo de respuesta ──
   730→   730→    let objective: ResponseObjective = matchedPattern?.objective || 'dar_paso_concreto';
   731→   731→
   732→   732→    // Ajustes finos
   733→   733→    if (isRepetition) objective = 'ajustar_rumbo';
   734→   734→    if (isHybrid && primaryNeed === 'progreso' && secondaryNeed === 'descarga_emocional') {
   735→   735→      objective = 'validar_y_avanzar';
   736→   736→    }
   737→   737→
   738→   738→    // ── Paso 12: Confidence ──
   739→   739→    const confidence = computeConfidence(!!matchedPattern, domains.length, isHybrid);
   740→   740→
   741→   741→    // ── Paso 13: FREE simplificación ──
   742→   742→    if (!isPremium) {
   743→   743→      // FREE: reducir sistemas relevantes al máximo 3
   744→   744→      if (relevant.length > 3) {
   745→   745→        relevant.length = 3;
   746→   746→      }
   747→   747→      // FREE: nunca desafiar
   748→   748→      if (shouldChallenge) {
   749→   749→        // No hay campo mutable, se refleja en el output
   750→   750→      }
   751→   751→    }
   752→   752→
   753→   753→    // ── Construir intent descriptivo ──
   754→   754→    const intentMap: Record<PrimaryNeed, string> = {
   755→   755→      orientacion: 'solicitar_orientacion',
   756→   756→      descarga_emocional: 'expresar_dolor',
   757→   757→      progreso: 'compartir_progreso',
   758→   758→      reflexion: 'cuestionar_situacion',
   759→   759→      informacion: 'pedir_informacion',
   760→   760→      celebracion: 'celebrar_logro',
   761→   761→      desafio: 'expresar_conformidad',
   762→   762→      continuidad: 'retomar_hilo_anterior',
   763→   763→      urgencia: 'pedir_ayuda_inmediata',
   764→   764→      exploracion: 'explorar_possibilidades',
   765→   765→    };
   766→   766→
   767→   767→    return {
   768→   768→      primaryNeed,
   769→   769→      secondaryNeed,
   770→   770→      mainIntent: matchedPattern?.intent || intentMap[primaryNeed] || 'general',
   771→   771→      secondaryIntent: secondaryNeed ? intentMap[secondaryNeed] || null : null,
   772→   772→      priorityDomain,
   773→   773→      secondaryDomain,
   774→   774→      reasoningDepth: depth,
   775→   775→      responseObjective: objective,
   776→   776→      recommendedTone: tone,
   777→   777→      recommendedLength: length,
   778→   778→      relevantSystems: relevant,
   779→   779→      ignoredSystems: ignored,
   780→   780→      askQuestion: ask,
   781→   781→      maxQuestions: max,
   782→   782→      shouldCelebrate,
   783→   783→      shouldChallenge,
   784→   784→      shouldReflect,
   785→   785→      shouldBePractical,
   786→   786→      confidence,
   787→   787→    };
   788→   788→  } catch {
   789→   789→    return null;
   790→   790→  }
   791→   791→}
