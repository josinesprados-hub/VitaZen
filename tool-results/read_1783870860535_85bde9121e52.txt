    55→/** Icon placeholder — mimics the icon-sm / icon-md / icon-lg containers */
    56→export function SkeletonIcon({ size = 'md', className }: { size?: 'sm' | 'md' | 'lg'; className?: string }) {
    57→  const sizeClasses = {
    58→    sm: 'w-9 h-9 rounded-lg',
    59→    md: 'w-10 h-10 rounded-lg',
    60→    lg: 'w-12 h-12 rounded-xl',
    61→  };
    62→  return <div className={cn('premium-shimmer', sizeClasses[size], className)} />;
    63→}
