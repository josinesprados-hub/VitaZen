    87→function getRelativeDate(dateStr: string): string {
    88→  const date = new Date(dateStr);
    89→  const now = new Date();
    90→  const diffMs = now.getTime() - date.getTime();
    91→  const diffMins = Math.floor(diffMs / 60000);
    92→  const diffHours = Math.floor(diffMs / 3600000);
    93→
    94→  // Calendar-day difference using Madrid-normalized dates
    95→  const entryKey = getMadridDateKey(date);
    96→  const todayKey = getTodayDateKey();
    97→  const [eY, eM, eD] = entryKey.split('-').map(Number);
    98→  const [tY, tM, tD] = todayKey.split('-').map(Number);
    99→  const entryDate = new Date(eY, eM - 1, eD);
   100→  const todayDate = new Date(tY, tM - 1, tD);
   101→  const diffDays = Math.round((todayDate.getTime() - entryDate.getTime()) / 86400000);
   102→
   103→  if (diffMins < 1 && diffDays === 0) return 'Ahora';
   104→  if (diffMins < 60 && diffDays === 0) return `Hace ${diffMins} min`;
   105→  if (diffHours < 24 && diffDays === 0) return `Hace ${diffHours}h`;
   106→  if (diffDays === 1) return 'Ayer';
   107→  if (diffDays < 7) return `Hace ${diffDays} días`;
   108→  if (diffDays < 30) {
   109→    const weeks = Math.floor(diffDays / 7);
   110→    return weeks === 1 ? 'Hace 1 semana' : `Hace ${weeks} semanas`;
   111→  }
   112→  return entryDate.toLocaleDateString('es-ES', { day: 'numeric', month: 'short' });
   113→}

   115→function getDateGroup(dateStr: string): string {
   116→  // Calendar-day difference using Madrid-normalized dates
   117→  const entryKey = getMadridDateKey(new Date(dateStr));
   118→  const todayKey = getTodayDateKey();
   119→
   120→  if (entryKey === todayKey) return 'Hoy';
   121→
   122→  const [eY, eM, eD] = entryKey.split('-').map(Number);
   123→  const [tY, tM, tD] = todayKey.split('-').map(Number);
   124→  const entryDate = new Date(eY, eM - 1, eD);
   125→  const todayDate = new Date(tY, tM - 1, tD);
   126→  const diffDays = Math.round((todayDate.getTime() - entryDate.getTime()) / 86400000);
   127→
   128→  if (diffDays === 1) return 'Ayer';
   129→  if (diffDays < 7) return 'Esta semana';
   130→  if (diffDays < 30) return 'Este mes';
   131→  return 'Anterior';
   132→}

   151→function getProgressColor(rem: number, limit: number): string {
   152→  if (!isFinite(limit)) return '#c8a55a';
   153→  const pct = rem / limit;
   154→  if (pct > 0.5) return '#c8a55a';
   155→  if (pct > 0.25) return '#e8a849';
   156→  return '#ef4444';

   537→    const userMessage: Message = {
   538→      id: 'temp-' + Date.now(),
   539→      role: 'user',

   541→      createdAt: new Date().toISOString(),

   546→
   547→    try {
   548→      const res = await apiFetch('/api/ai/chat', {
   549→        method: 'POST',
   550→        body: JSON.stringify({ threadId: sentThreadId, content }),
   551→      });

   553→      if (res.status === 403) {
   554→        const data = await res.json();

   556→        setRemaining(0);

   560→        // L-4 FIX: Restore the input text so the user doesn't lose their message.
   561→        // Previously the input was cleared (line 547) and never restored on 403.

   565→
   566→      if (res.ok) {
