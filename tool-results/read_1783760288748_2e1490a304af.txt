   735→   735→function daysAgo(date: Date): number {
   736→   736→  const todayKey = getTodayDateKey();
   737→   737→  const dateKey = getMadridDateKey(date);

   739→   739→  const dateMs = new Date(dateKey + 'T00:00:00').getTime();
   740→   740→  return Math.round((todayMs - dateMs) / 86400000);
   741→   741→}

   748→   748→/**
   749→   749→ * Extract a fragment from journal content for the mentor context.
   750→   750→ * Returns null if the content doesn't add information beyond the title.
   751→   751→ *
   752→   752→ * Rules:
   753→   753→ * - Max 180 characters
   754→   754→ * - Never cut words
   755→   755→ * - Remove line breaks and repeated spaces
   756→   756→ * - End with "…" when truncated
   757→   757→ * - No summarization, no reinterpretation — exact user text only
   758→   758→ * - Skip if content only repeats what the title already says
   759→   759→ */
   760→   760→function extractJournalFragment(title: string, content: string): string | null {
   761→   761→  // Clean content: remove line breaks, collapse repeated spaces, trim
   762→   762→  const cleaned = content
   763→   763→    .replace(/\n/g, ' ')
   764→   764→    .replace(/\s+/g, ' ')
   765→   765→    .trim();
   766→   766→
   767→   767→  if (!cleaned) return null;
   768→   768→
   769→   769→  // Normalize title for comparison (lowercase, no punctuation, collapsed spaces)
   770→   770→  const normalizeForComparison = (s: string) =>
   771→   771→    s.toLowerCase().replace(/[^\wáéíóúñü]/g, ' ').replace(/\s+/g, ' ').trim();
   772→   772→
   773→   773→  const normalizedTitle = normalizeForComparison(title);
   774→   774→  const normalizedCleaned = normalizeForComparison(cleaned);
   775→   775→
   776→   776→  // If content starts with the title (user often repeats title as first line),
   777→   777→  // strip the title prefix from content for comparison
   778→   778→  let contentForComparison = normalizedCleaned;
   779→   779→  if (normalizedCleaned.startsWith(normalizedTitle)) {
   780→   780→    contentForComparison = normalizedCleaned.slice(normalizedTitle.length).trim();
   781→   781→  }
   782→   782→
   783→   783→  // If what remains is empty or too short, content doesn't add information
   784→   784→  if (contentForComparison.length < 10) return null;
   785→   785→
   786→   786→  // Truncate to 180 chars without cutting words
   787→   787→  const MAX_LEN = 180;
   788→   788→  if (cleaned.length <= MAX_LEN) return cleaned;
   789→   789→
   790→   790→  // Find the last space before the limit
   791→   791→  const truncated = cleaned.slice(0, MAX_LEN);
   792→   792→  const lastSpace = truncated.lastIndexOf(' ');
   793→   793→  if (lastSpace <= 0) return cleaned.slice(0, MAX_LEN) + '…';
   794→   794→
   795→   795→  return truncated.slice(0, lastSpace) + '…';
   796→   796→}

   801→   801→function formatBasicContext(ctx: UserContext): string {

   824→   824→    const last = ctx.recentMeditations[0];
   825→   825→    const days = daysAgo(last.completedAt);

   833→   833→    const title = last.title || 'sin título';
   834→   834→    const days = daysAgo(last.createdAt);

   839→   839→  // Consistency signal

   841→   841→  if (c.trend === 'improving') {
   842→   842→    lines.push(`Viene siendo más constante esta semana.`);
   843→   843→  } else if (c.trend === 'declining') {
   844→   844→    lines.push(`Últimamente menos activo/a que antes.`);
   845→   845→  }
   846→   846→
   847→   847→  return lines.join('\n');

  1348→  1348→}
