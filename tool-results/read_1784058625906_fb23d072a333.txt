   169→/**
   170→ * Classify the user message into relevance domains.
   171→ * Returns domain signals sorted by strength (descending).
   172→ * Pure keyword matching — zero latency.
   173→ */
   174→function classifyDomain(userMessage: string): DomainSignal[] {
   175→  const msg = userMessage.toLowerCase();
   176→  const signals: DomainSignal[] = [];
   177→
   178→  for (const rule of DOMAIN_RULES) {
   179→    let matchCount = 0;
   180→    for (const kw of rule.keywords) {
   181→      if (kw.test(msg)) matchCount++;
   182→    }
   183→
   184→    if (matchCount > 0) {
   185→      // Strength = boost + (matches * 15), capped at 100
   186→      const strength = Math.min(rule.boost + matchCount * 15, 100);
   187→      signals.push({ domain: rule.domain, strength });
   188→    }
   189→  }
   190→
   191→  // Sort by strength descending
   192→  signals.sort((a, b) => b.strength - a.strength);
   193→  return signals;
   194→}

   306→/**
   307→ * Score a context block based on domain signals.
   308→ * Weighted average across all detected domains.
   309→ */
   310→function scoreBlock(
   311→  blockId: ContextBlockId,
   312→  domains: DomainSignal[],
   313→  totalDomainStrength: number
   314→): number {
   315→  if (totalDomainStrength === 0) return 50; // neutral default
   316→
   317→  let weightedScore = 0;
   318→  let hasOpinion = false;
   319→
   320→  for (const signal of domains) {
   321→    const matrix = RELEVANCE_MATRIX[signal.domain];
   322→    if (!matrix) continue;
   323→
   324→    const weight = matrix[blockId];
   325→    if (weight === undefined) continue;
   326→
   327→    // Weight the matrix score by the domain's strength
   328→    weightedScore += weight * (signal.strength / 100);
   329→    hasOpinion = true;
   330→  }
   331→
   332→  if (!hasOpinion) return 50; // no domain has an opinion about this block
   333→
   334→  // Normalize: weighted average
   335→  const avgStrength = domains.reduce((s, d) => s + d.strength, 0) / domains.length;
   336→  return Math.round(weightedScore / (avgStrength / 100));
   337→}

   453→/**
   454→ * Parse the assembled system prompt into identifiable blocks.
   455→ *
   456→ * Strategy:
   457→ * 1. The base prompt (everything before CONTEXT_START_MARKER) is always protected.
   458→ * 2. The context section (between markers) is split into blocks.
   459→ * 3. Everything after CONTEXT_END_MARKER (rules, understanding) is separate.
   460→ */
   461→function parseBlocks(systemPrompt: string): {
   462→  basePrompt: string;
   463→  contextBlocks: ContextBlock[];
   464→  rulesBlock: string;
   465→  understandingBlock: string;
   466→} {
   467→  const startIdx = systemPrompt.indexOf(CONTEXT_START_MARKER);
   468→  const endIdx = systemPrompt.indexOf(CONTEXT_END_MARKER);
   469→
   470→  // Case 1: No context markers at all — return as-is
   471→  if (startIdx === -1 || endIdx === -1) {
   472→    // Check if there's an understanding block appended after the base prompt
   473→    const understandingIdx = systemPrompt.lastIndexOf('\n\n');
   474→
   475→    return {
   476→      basePrompt: systemPrompt,
   477→      contextBlocks: [],
   478→      rulesBlock: '',
   479→      understandingBlock: '',
   480→    };
   481→  }
   482→
   483→  // Extract sections
   484→  const basePrompt = systemPrompt.substring(0, startIdx).trim();
   485→  const contextSection = systemPrompt.substring(
   486→    startIdx + CONTEXT_START_MARKER.length,
   487→    endIdx
   488→  ).trim();
   489→  const afterContext = systemPrompt.substring(endIdx + CONTEXT_END_MARKER.length).trim();
   490→
   491→  // Split context section into lines and assign to blocks
   492→  const lines = contextSection.split('\n');
   493→  const blockTexts: Partial<Record<ContextBlockId, string[]>> = {};
   494→  const unassignedLines: string[] = [];
   495→
   496→  for (const line of lines) {
   497→    const trimmed = line.trim();
   498→    if (!trimmed) continue;
   499→
   500→    let assigned = false;
   501→    for (const indicator of BLOCK_INDICATORS) {
   502→      if (indicator.patterns.some(p => p.test(trimmed))) {
   503→        if (!blockTexts[indicator.id]) blockTexts[indicator.id] = [];
   504→        blockTexts[indicator.id]!.push(trimmed);
   505→        assigned = true;
   506→        break;
   507→      }
   508→    }
   509→
   510→    if (!assigned) {
   511→      // Try to attach to the most recent block
   512→      unassignedLines.push(trimmed);
   513→    }
   514→  }
   515→
   516→  // Attach unassigned lines to the most recent preceding block
   517→  if (unassignedLines.length > 0) {
   518→    // Find the last block that was assigned
   519→    const blockOrder: ContextBlockId[] = [
   520→      'identity', 'emotional_state', 'life_stage', 'patterns',
   521→      'silent_memories', 'lived_experience', 'behavioral', 'conversational',
   522→    ];
   523→
   524→    let lastBlock: ContextBlockId | null = null;
   525→    for (let i = lines.length - 1; i >= 0; i--) {
   526→      const trimmed = lines[i].trim();
   527→      if (!trimmed) continue;
   528→      for (const indicator of BLOCK_INDICATORS) {
   529→        if (indicator.patterns.some(p => p.test(trimmed))) {
   530→          lastBlock = indicator.id;
   531→          break;
   532→        }
   533→      }
   534→      if (lastBlock) break;
   535→    }
   536→
   537→    if (lastBlock && blockTexts[lastBlock]) {
   538→      blockTexts[lastBlock]!.push(...unassignedLines);
   539→    }
   540→  }
   541→
   542→  // Build context blocks
   543→  const contextBlocks: ContextBlock[] = Object.entries(blockTexts).map(([id, texts]) => ({
   544→    id: id as ContextBlockId,
   545→    text: texts!.join('\n'),
   546→    charCount: texts!.join('\n').length,
   547→    protected: false,
   548→    relevance: 0, // scored later
   549→    budget: 0,   // set later
   550→  }));
   551→
   552→  // Split afterContext into rules and understanding
   553→  // Rules contain "CÓMO USAR ESTE CONTEXTO" and "CONTROL DE EVIDENCIA"
   554→  let rulesBlock = '';
   555→  let understandingBlock = '';
   556→
   557→  // The understanding block was injected after the context block
   558→  // in route.ts: systemPrompt = systemPrompt + '\n\n' + adaptationSnippet
   559→  // So it's the last part of afterContext
   560→  const rulesEndMarker = 'CONTROL DE EVIDENCIA:';
   561→  const rulesEndIdx = afterContext.indexOf(rulesEndMarker);
   562→
   563→  if (rulesEndIdx !== -1) {
   564→    // There's a "CONTROL DE EVIDENCIA" section (PREMIUM)
   565→    const afterEvidence = afterContext.substring(rulesEndIdx + rulesEndMarker.length);
   566→    // The understanding block comes after the rules — it's the last \n\n separated block
   567→    // that doesn't contain "CÓMO USAR" or "CONTROL DE EVIDENCIA"
   568→    const rulesPart = afterContext.substring(0, rulesEndIdx + rulesEndMarker.length);
   569→    rulesBlock = rulesPart;
   570→
   571→    // Understanding is everything after the evidence rules that looks like adaptation
   572→    const evidenceEnd = afterEvidence.indexOf('\n\n');
   573→    if (evidenceEnd !== -1) {
   574→      const afterEvidenceRules = afterEvidence.substring(evidenceEnd + 2).trim();
   575→      if (afterEvidenceRules.length > 0 && !afterEvidenceRules.includes('CÓMO USAR')) {
   576→        understandingBlock = afterEvidenceRules;
   577→      }
   578→    }
   579→  } else {
   580→    // No CONTROL DE EVIDENCIA — could be FREE or PREMIUM without that section
   581→    // Check if there's content after "CÓMO USAR ESTE CONTEXTO" rules
   582→    const comoUsarEnd = afterContext.indexOf('CÓMO USAR ESTE CONTEXTO:');
   583→    if (comoUsarEnd !== -1) {
   584→      // Find the end of the rules section (next double newline after rules)
   585→      const rulesSection = afterContext.substring(comoUsarEnd);
   586→      // Rules end at the next section or at the understanding block
   587→      const potentialUnderstanding = afterContext.substring(comoUsarEnd).split('\n\n').pop()?.trim() || '';
   588→      // The understanding block is the last paragraph if it doesn't contain rule keywords
   589→      if (
   590→        potentialUnderstanding.length > 0 &&
   591→        !potentialUnderstanding.includes('CÓMO USAR') &&
   592→        !potentialUnderstanding.includes('integrar de forma invisible') &&
   593→        !potentialUnderstanding.includes('máximo UNA referencia')
   594→      ) {
   595→        rulesBlock = afterContext.substring(0, afterContext.lastIndexOf(potentialUnderstanding)).trim();
   596→        understandingBlock = potentialUnderstanding;
   597→      } else {
   598→        rulesBlock = afterContext;
   599→      }
   600→    } else {
   601→      rulesBlock = afterContext;
   602→    }
   603→  }
   604→
   605→  return { basePrompt, contextBlocks, rulesBlock, understandingBlock };
   606→}
   607→
   608→// ═══════════════════════════════════════════
   609→// 5. CONFLICT RESOLUTION
   610→// When two blocks propose incompatible
   611→// approaches, the higher-relevance block wins.
   612→// ═══════════════════════════════════════════
   613→
   614→/**
   615→ * Known conflict pairs. When both blocks are active and their
   616→ * signals conflict, the block with higher relevance wins and
   617→ * the other gets its relevance reduced.
   618→ */
   619→interface ConflictRule {
   620→  blockA: ContextBlockId;
   621→  blockB: ContextBlockId;
   622→  /** When A's relevance is > threshold AND B's relevance is < threshold, B is suppressed */
   623→  aDominanceThreshold: number;
   624→  /** How much to reduce B's relevance when suppressed */
   625→  suppressionPenalty: number;
   626→  /** Condition: which domain must be dominant for this conflict to apply */
   627→  triggerDomain: RelevanceDomain;
   628→}
   629→
   630→const CONFLICT_RULES: ConflictRule[] = [
   631→  {
   632→    // When user is in crisis, behavioral/push context is counterproductive
   633→    blockA: 'emotional_state',
   634→    blockB: 'behavioral',
   635→    aDominanceThreshold: 70,
   636→    suppressionPenalty: 60,
   637→    triggerDomain: 'crisis',
   638→  },
   639→  {
   640→    // When user talks about emotions, detailed financial context is noise
   641→    blockA: 'understanding',
   642→    blockB: 'patterns',
   643→    aDominanceThreshold: 70,
   644→    suppressionPenalty: 50,
   645→    triggerDomain: 'emotional',
   646→  },
   647→  {
   648→    // When user talks about practical steps, silent memories add no value
   649→    blockA: 'behavioral',
   650→    blockB: 'silent_memories',
   651→    aDominanceThreshold: 70,
   652→    suppressionPenalty: 50,
   653→    triggerDomain: 'practical',
   654→  },
   655→];
   656→
   657→function resolveConflicts(
   658→  blocks: ContextBlock[],
   659→  domains: DomainSignal[]
   660→): ContextBlock[] {
   661→  const dominantDomain = domains.length > 0 ? domains[0].domain : null;
   662→
   663→  for (const rule of CONFLICT_RULES) {
   664→    // Only apply if the dominant domain matches
   665→    if (!dominantDomain || dominantDomain !== rule.triggerDomain) continue;
   666→
   667→    const blockA = blocks.find(b => b.id === rule.blockA);
   668→    const blockB = blocks.find(b => b.id === rule.blockB);
   669→
   670→    if (!blockA || !blockB) continue;
   671→
   672→    if (blockA.relevance >= rule.aDominanceThreshold) {
   673→      blockB.relevance = Math.max(0, blockB.relevance - rule.suppressionPenalty);
   674→    }
   675→  }
   676→
   677→  return blocks;
   678→}
   679→
   680→// ═══════════════════════════════════════════
   681→// 6. BLOCK TRUNCATION
   682→// When a block exceeds its budget,
   683→// truncate intelligently (keep first lines).
   684→// ═══════════════════════════════════════════
   685→
   686→function truncateBlock(block: ContextBlock, maxChars: number): ContextBlock {
   687→  if (block.charCount <= maxChars) return block;
   688→
   689→  const lines = block.text.split('\n');
   690→  let totalChars = 0;
   691→  const keptLines: string[] = [];
   692→
   693→  for (const line of lines) {
   694→    if (totalChars + line.length > maxChars) break;
   695→    keptLines.push(line);
   696→    totalChars += line.length + 1; // +1 for newline
   697→  }
   698→
   699→  return {
   700→    ...block,
   701→    text: keptLines.join('\n'),
   702→    charCount: totalChars,
   703→  };
   704→}
   705→
   706→// ═══════════════════════════════════════════
   707→// 7. MAIN ENTRY POINT
   708→// The only exported function. Called once
   709→// per message in route.ts.
   710→// ═══════════════════════════════════════════
   711→
   712→/**
   713→ * Optimize the assembled system prompt for the current user message.
   714→ *
   715→ * This is the Decision Engine's sole public function.
   716→ * It takes the already-assembled system prompt and the user's message,
   717→ * then returns an optimized version that uses only the most relevant
   718→ * context blocks.
   719→ *
   720→ * Integration point: route.ts, AFTER buildContextualSystemPrompt
   721→ * and getUnderstandingContext, BEFORE assembling groqMessages.
   722→ *
   723→ * @param systemPrompt - The fully assembled system prompt
   724→ * @param userMessage - The current user message
   725→ * @param plan - User's plan ('FREE' or 'PREMIUM')
   726→ * @returns Optimized system prompt + decision metadata
   727→ */
   728→export function optimizeContext(
   729→  systemPrompt: string,
   730→  userMessage: string,
   731→  plan: string
   732→): DecisionResult {
   733→  const budget = BUDGETS[plan] || BUDGETS.FREE;
   734→
   735→  // Step 1: Parse the prompt into blocks
   736→  const { basePrompt, contextBlocks, rulesBlock, understandingBlock } = parseBlocks(systemPrompt);
   737→
   738→  // If no context blocks exist, return as-is
   739→  if (contextBlocks.length === 0) {
   740→    return {
   741→      systemPrompt,
   742→      activeBlocks: ['base_prompt'],
   743→      filteredBlocks: [],
   744→      truncatedBlocks: [],
   745→      charsSaved: 0,
   746→    };
   747→  }
   748→
   749→  // Step 2: Classify user message into domains
   750→  const domains = classifyDomain(userMessage);
   751→  const totalDomainStrength = domains.reduce((s, d) => s + d.strength, 0);
   752→
   753→  // Step 3: Score each block
   754→  const scoredBlocks = contextBlocks.map(block => ({
   755→    ...block,
   756→    relevance: scoreBlock(block.id, domains, totalDomainStrength),
   757→    budget: budget.blockMax[block.id] || budget.maxContextChars,
   758→  }));
   759→
   760→  // Step 4: Resolve conflicts between blocks
   761→  const resolvedBlocks = resolveConflicts(scoredBlocks, domains);
   762→
   763→  // Step 5: Filter — remove blocks with negative relevance, respect maxActiveBlocks
   764→  // Sort by relevance descending
   765→  const sorted = [...resolvedBlocks].sort((a, b) => b.relevance - a.relevance);
   766→
   767→  // Separate positive-relevance blocks from negative
   768→  const positive = sorted.filter(b => b.relevance > 0);
   769→  const negative = sorted.filter(b => b.relevance <= 0);
   770→
   771→  // Keep top N blocks by relevance
   772→  const kept = positive.slice(0, budget.maxActiveBlocks);
   773→
   774→  // Track filtered blocks
   775→  const filteredIds: ContextBlockId[] = [
   776→    ...negative.map(b => b.id),
   777→    ...positive.slice(budget.maxActiveBlocks).map(b => b.id),
   778→  ];
   779→
   780→  // Step 6: Truncate kept blocks to their budget
   781→  const truncatedIds: ContextBlockId[] = [];
   782→  const finalBlocks = kept.map(block => {
   783→    const truncated = truncateBlock(block, block.budget);
   784→    if (truncated.charCount < block.charCount) {
   785→      truncatedIds.push(block.id);
   786→    }
   787→    return truncated;
   788→  });
   789→
   790→  // Step 7: Apply global context budget
   791→  let totalContextChars = finalBlocks.reduce((s, b) => s + b.charCount, 0);
   792→  if (totalContextChars > budget.maxContextChars) {
   793→    // Remove lowest-relevance blocks until within budget
   794→    finalBlocks.sort((a, b) => b.relevance - a.relevance);
   795→    while (totalContextChars > budget.maxContextChars && finalBlocks.length > 1) {
   796→      const removed = finalBlocks.pop()!;
   797→      filteredIds.push(removed.id);
   798→      totalContextChars -= removed.charCount;
   799→    }
   800→  }
   801→
   802→  // Step 8: Handle understanding block
   803→  let finalUnderstanding = understandingBlock;
   804→  const understandingBudget = budget.blockMax.understanding || 150;
   805→  if (understandingBlock.length > understandingBudget) {
   806→    // Truncate understanding to first N lines that fit
   807→    const lines = understandingBlock.split('\n');
   808→    let chars = 0;
   809→    const keptLines: string[] = [];
   810→    for (const line of lines) {
   811→      if (chars + line.length > understandingBudget) break;
   812→      keptLines.push(line);
   813→      chars += line.length + 1;
   814→    }
   815→    finalUnderstanding = keptLines.join('\n');
   816→    if (finalUnderstanding.length < understandingBlock.length) {
   817→      truncatedIds.push('understanding');
   818→    }
   819→  }
   820→
   821→  // Step 9: Reassemble the system prompt
   822→  const activeIds: ContextBlockId[] = [...finalBlocks.map(b => b.id)];
   823→  if (finalUnderstanding) activeIds.push('understanding');
   824→  activeIds.push('base_prompt');
   825→
   826→  const contextText = finalBlocks.map(b => b.text).join('\n');
   827→
   828→  let finalPrompt: string;
   829→  if (contextText.trim().length > 0) {
   830→    finalPrompt = `${basePrompt}\n\n── Lo que sabes de esta persona ──\n${contextText.trim()}\n── Fin ──`;
   831→    if (rulesBlock) {
   832→      finalPrompt += `\n\n${rulesBlock}`;
   833→    }
   834→    if (finalUnderstanding.trim()) {
   835→      finalPrompt += `\n\n${finalUnderstanding.trim()}`;
   836→    }
   837→  } else {
   838→    // No context survived filtering — use base prompt only
   839→    finalPrompt = basePrompt;
   840→    if (rulesBlock) {
   841→      // Keep a minimal version of rules
   842→      finalPrompt += `\n\n${rulesBlock}`;
   843→    }
   844→  }
   845→
   846→  // Calculate chars saved
   847→  const originalContextChars = contextBlocks.reduce((s, b) => s + b.charCount, 0) + understandingBlock.length;
   848→  const finalContextChars = finalBlocks.reduce((s, b) => s + b.charCount, 0) + finalUnderstanding.length;
   849→  const charsSaved = Math.max(0, originalContextChars - finalContextChars);
   850→
   851→  return {
   852→    systemPrompt: finalPrompt,
   853→    activeBlocks: activeIds,
   854→    filteredBlocks: filteredIds,
   855→    truncatedBlocks: truncatedIds,
   856→    charsSaved,
   857→  };
   858→}
