   106→   106→function TrendBadge({ value, invert = false }: { value: number; invert?: boolean }) {
   107→   107→  // invert=true means positive when value is negative (e.g. stress going down)
   108→   108→  const positive = invert ? value < -0.2 : value > 0.2;
   109→   109→  const negative = invert ? value > 0.2 : value < -0.2;
   110→   110→
   111→   111→  if (positive) {
   112→   112→    return (
   113→   113→      <span className="flex items-center gap-1 text-[11px] text-[#22c55e]">
   114→   114→        <TrendingUp size={11} /> {invert ? '↓' : '↑'}
   115→   115→      </span>
   116→   116→    );
   117→   117→  }
   118→   118→  if (negative) {
   119→   119→    return (
   120→   120→      <span className="flex items-center gap-1 text-[11px] text-champagne-warm">
   121→   121→        <TrendingDown size={11} /> {invert ? '↑' : '↓'}
   122→   122→      </span>
   123→   123→    );
   124→   124→  }
   125→   125→  return (
   126→   126→    <span className="flex items-center gap-1 text-[11px] text-[#555]">
   127→   127→      <Minus size={11} /> —
   128→   128→    </span>
   129→   129→  );
   130→   130→}
