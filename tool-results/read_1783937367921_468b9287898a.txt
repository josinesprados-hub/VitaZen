   112→   112→function parseQuickCapture(input: string, historyMap: Record<string, string>): ParsedCapture | null {
   113→   113→  if (!input.trim()) return null;
   114→   114→
   115→   115→  const tokens = input.trim().split(/\s+/);
   116→   116→  let amount = 0;
   117→   117→  let descriptionTokens: string[] = [];
   118→   118→  let amountFound = false;
   119→   119→
   120→   120→  for (let i = tokens.length - 1; i >= 0; i--) {
   121→   121→    if (/^[\d.,]+$/.test(tokens[i])) {
   122→   122→      const parsed = parseEuropeanQuick(tokens[i]);
   123→   123→      if (parsed > 0) {
   124→   124→        amount = parsed;
   125→   125→        descriptionTokens = tokens.slice(0, i);
   126→   126→        amountFound = true;
   127→   127→        break;
   128→   128→      }
   129→   129→    }
   130→   130→  }
   131→   131→
   132→   132→  if (!amountFound && /^[\d.,]+$/.test(tokens[0])) {
   133→   133→    const parsed = parseEuropeanQuick(tokens[0]);
   134→   134→    if (parsed > 0) {
   135→   135→      amount = parsed;
   136→   136→      descriptionTokens = tokens.slice(1);
   137→   137→      amountFound = true;
   138→   138→    }
   139→   139→  }
   140→   140→
   141→   141→  if (!amountFound) {
   142→   142→    const match = input.match(/([\d.,]+)\s*€?\s*$/);
   143→   143→    if (match) {
   144→   144→      const parsed = parseEuropeanQuick(match[1]);
   145→   145→      if (parsed > 0) {
   146→   146→        amount = parsed;
   147→   147→        descriptionTokens = [input.replace(match[0], '').trim()];
   148→   148→        amountFound = true;
   149→   149→      }
   150→   150→    }
   151→   151→  }
   152→   152→
   153→   153→  if (!amountFound || amount <= 0) return null;
   154→   154→
   155→   155→  const description = descriptionTokens.join(' ').trim();
   156→   156→  const textToMatch = description || input;
   157→   157→
   158→   158→  let category = '';
   159→   159→  const descLower = textToMatch.toLowerCase();
   160→   160→  if (historyMap[descLower]) {
   161→   161→    category = historyMap[descLower];
   162→   162→  }
   163→   163→
   164→   164→  if (!category) {
   165→   165→    for (const [regex, cat] of KEYWORD_CATEGORIES) {
   166→   166→      if (regex.test(textToMatch)) {
   167→   167→        category = cat;
   168→   168→        break;
   169→   169→      }
   170→   170→    }
   171→   171→  }
   172→   172→
   173→   173→  if (!category) category = 'Otros';
   174→   174→
   175→   175→  const type: 'expense' | 'income' = INCOME_KEYWORDS.test(textToMatch) ? 'income' : 'expense';
   176→   176→
   177→   177→  return {
   178→   178→    description: description || category,
   179→   179→    amount,
   180→   180→    category,
   181→   181→    type,
   182→   182→  };
   183→   183→}
   184→   184→
   185→   185→function parseEuropeanQuick(raw: string): number {
   186→   186→  const trimmed = raw.trim();
   187→   187→  if (!trimmed) return 0;
   188→   188→  const hasComma = trimmed.includes(',');
   189→   189→  const hasDot = trimmed.includes('.');
   190→   190→  let normalized: string;
   191→   191→
   192→   192→  if (hasComma && hasDot) {
   193→   193→    normalized = trimmed.replace(/\./g, '').replace(',', '.');
   194→   194→  } else if (hasComma) {
   195→   195→    normalized = trimmed.replace(',', '.');
   196→   196→  } else if (hasDot) {
   197→   197→    normalized = /^\d{1,3}(\.\d{3})+$/.test(trimmed)
   198→   198→      ? trimmed.replace(/\./g, '')
   199→   199→      : trimmed;
   200→   200→  } else {
   201→   201→    normalized = trimmed;
   202→   202→  }
   203→   203→
   204→   204→  normalized = normalized.replace(/\.$/, '');
   205→   205→  const parsed = parseFloat(normalized);
   206→   206→  return isNaN(parsed) ? 0 : parsed;
   207→   207→}

   433→   433→            </span>
   434→   434→          </div>
   435→   435→          <button
   436→   436→            onClick={handleSubmit}
   437→   437→            disabled={submitting}
   438→   438→            className="bg-champagne text-black px-3 py-1 rounded-lg text-xs font-semibold hover:bg-champagne-hover active:scale-95 transition-all disabled:opacity-50"
   439→   439→          >
