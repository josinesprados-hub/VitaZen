    68→    68→// Relative date helper
    69→    69→// ─────────────────────────────────────────
    70→    70→
    71→    71→function getRelativeDate(dateStr: string): string {
    72→    72→  const date = new Date(dateStr);
    73→    73→  const now = new Date();
    74→    74→  const diffMs = now.getTime() - date.getTime();
    75→    75→  const diffMins = Math.floor(diffMs / 60000);
    76→    76→  const diffHours = Math.floor(diffMs / 3600000);
    77→    77→
    78→    78→  // Calendar-day difference using Madrid-normalized dates
    79→    79→  const entryKey = getMadridDateKey(date);
    80→    80→  const todayKey = getTodayDateKey();
    81→    81→  const diffDays = daysBetweenDateKeys(entryKey, todayKey);
    82→    82→
    83→    83→  if (diffMins < 1 && diffDays === 0) return 'Ahora';
    84→    84→  if (diffMins < 60 && diffDays === 0) return `Hace ${diffMins} min`;
    85→    85→  if (diffHours < 24 && diffDays === 0) return `Hace ${diffHours}h`;
    86→    86→  if (diffDays === 1) return 'Ayer';
    87→    87→  if (diffDays < 7) return `Hace ${diffDays} días`;
    88→    88→  if (diffDays < 30) {
    89→    89→    const weeks = Math.floor(diffDays / 7);
    90→    90→    return weeks === 1 ? 'Hace 1 semana' : `Hace ${weeks} semanas`;
    91→    91→  }
    92→    92→  const [eY, eM, eD] = entryKey.split('-').map(Number);
    93→    93→  const entryDate = new Date(eY, eM - 1, eD);
    94→    94→  return entryDate.toLocaleDateString('es-ES', { day: 'numeric', month: 'short' });
    95→    95→}

    97→    97→function getDateGroup(dateStr: string): string {
    98→    98→  // Calendar-day difference using Madrid-normalized dates
    99→    99→  const entryKey = getMadridDateKey(new Date(dateStr));
   100→   100→  const todayKey = getTodayDateKey();
   101→   101→
   102→   102→  if (entryKey === todayKey) return 'Hoy';
   103→   103→
   104→   104→  const diffDays = daysBetweenDateKeys(entryKey, todayKey);
   105→   105→
   106→   106→  if (diffDays === 1) return 'Ayer';
   107→   107→  if (diffDays < 7) return 'Esta semana';
   108→   108→  if (diffDays < 30) return 'Este mes';
   109→   109→  return 'Anterior';
   110→   110→}
