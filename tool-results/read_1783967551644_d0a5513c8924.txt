    65→// ─────────────────────────────────────────
    66→
    67→function getRelativeDate(dateStr: string): string {
    68→  const date = new Date(dateStr);
    69→  const now = new Date();
    70→  const diffMs = now.getTime() - date.getTime();
    71→  const diffMins = Math.floor(diffMs / 60000);
    72→  const diffHours = Math.floor(diffMs / 3600000);
    73→
    74→  // Calendar-day difference using Madrid-normalized dates
    75→  const entryKey = getMadridDateKey(date);
    76→  const todayKey = getTodayDateKey();
    77→  const [eY, eM, eD] = entryKey.split('-').map(Number);
    78→  const [tY, tM, tD] = todayKey.split('-').map(Number);
    79→  const entryDate = new Date(eY, eM - 1, eD);
    80→  const todayDate = new Date(tY, tM - 1, tD);
    81→  const diffDays = Math.round((todayDate.getTime() - entryDate.getTime()) / 86400000);
    82→
    83→  if (diffMins < 1 && diffDays === 0) return 'Ahora';
    84→  if (diffMins < 60 && diffDays === 0) return `Hace ${diffMins} min`;
    85→  if (diffHours < 24 && diffDays === 0) return `Hace ${diffHours}h`;
    86→  if (diffDays === 1) return 'Ayer';
    87→  if (diffDays < 7) return `Hace ${diffDays} días`;
    88→  if (diffDays < 30) {
    89→    const weeks = Math.floor(diffDays / 7);
    90→    return weeks === 1 ? 'Hace 1 semana' : `Hace ${weeks} semanas`;
    91→  }
    92→  return entryDate.toLocaleDateString('es-ES', { day: 'numeric', month: 'short' });
    93→}

    95→function getDateGroup(dateStr: string): string {
    96→  // Calendar-day difference using Madrid-normalized dates
    97→  const entryKey = getMadridDateKey(new Date(dateStr));
    98→  const todayKey = getTodayDateKey();
    99→
   100→  if (entryKey === todayKey) return 'Hoy';
   101→
   102→  const [eY, eM, eD] = entryKey.split('-').map(Number);
   103→  const [tY, tM, tD] = todayKey.split('-').map(Number);
   104→  const entryDate = new Date(eY, eM - 1, eD);
   105→  const todayDate = new Date(tY, tM - 1, tD);
   106→  const diffDays = Math.round((todayDate.getTime() - entryDate.getTime()) / 86400000);
   107→
   108→  if (diffDays === 1) return 'Ayer';
   109→  if (diffDays < 7) return 'Esta semana';
   110→  if (diffDays < 30) return 'Este mes';
   111→  return 'Anterior';
   112→}

   132→  if (!isFinite(limit)) return '#c8a55a';
   133→  const pct = rem / limit;
   134→  if (pct > 0.5) return '#c8a55a';
   135→  if (pct > 0.25) return '#e8a849';
   136→  return '#ef4444';

   510→  const sendMessage = async () => {

   526→    };

   531→    try {
   532→      const res = await apiFetch('/api/ai/chat', {
   533→        method: 'POST',

   535→      });

   540→        setRemaining(0);
   541→        setDailyLimit(data.limit || 15);

   544→        // L-4 FIX: Restore the input text so the user doesn't lose their message.
   545→        // Previously the input was cleared (line 547) and never restored on 403.
   546→        setInput(content);
   547→        return;
   548→      }
   549→
   550→      if (res.ok) {
   551→        const data = await res.json();
   552→        const assistantMessage: Message = {
   553→          id: 'resp-' + Date.now(),
   554→          role: 'assistant',
   555→          content: data.message,
   556→          createdAt: new Date().toISOString(),
   557→        };
   558→        // Only add response if still on the same thread
   559→        if (activeThreadRef.current === sentThreadId) {
   560→          setMessages(prev => [...prev, assistantMessage]);
   561→          setRemaining(data.remaining);
   562→          setDailyLimit(data.limit || 15);
   563→          setIsContextual(!!data.contextual);
   564→        }
   565→
   566→        // Refresh threads to get updated title and updatedAt
