    91→function getRelativeDate(dateStr: string): string {
    92→  const date = new Date(dateStr);
    93→  const now = new Date();
    94→  const diffMs = now.getTime() - date.getTime();
    95→  const diffMins = Math.floor(diffMs / 60000);
    96→  const diffHours = Math.floor(diffMs / 3600000);
    97→
    98→  // Calendar-day difference using Madrid-normalized dates
    99→  const entryKey = getMadridDateKey(date);
   100→  const todayKey = getTodayDateKey();
   101→  const [eY, eM, eD] = entryKey.split('-').map(Number);
   102→  const [tY, tM, tD] = todayKey.split('-').map(Number);
   103→  const entryDate = new Date(eY, eM - 1, eD);
   104→  const todayDate = new Date(tY, tM - 1, tD);
   105→  const diffDays = Math.round((todayDate.getTime() - entryDate.getTime()) / 86400000);
   106→
   107→  if (diffMins < 1 && diffDays === 0) return 'Ahora';
   108→  if (diffMins < 60 && diffDays === 0) return `Hace ${diffMins} min`;
   109→  if (diffHours < 24 && diffDays === 0) return `Hace ${diffHours}h`;
   110→  if (diffDays === 1) return 'Ayer';
   111→  if (diffDays < 7) return `Hace ${diffDays} días`;
   112→  if (diffDays < 30) {
   113→    const weeks = Math.floor(diffDays / 7);
   114→    return weeks === 1 ? 'Hace 1 semana' : `Hace ${weeks} semanas`;
   115→  }
   116→  return entryDate.toLocaleDateString('es-ES', { day: 'numeric', month: 'short' });
   117→}

   119→function getDateGroup(dateStr: string): string {
   120→  // Calendar-day difference using Madrid-normalized dates
   121→  const entryKey = getMadridDateKey(new Date(dateStr));
   122→  const todayKey = getTodayDateKey();
   123→
   124→  if (entryKey === todayKey) return 'Hoy';
   125→
   126→  const [eY, eM, eD] = entryKey.split('-').map(Number);
   127→  const [tY, tM, tD] = todayKey.split('-').map(Number);
   128→  const entryDate = new Date(eY, eM - 1, eD);
   129→  const todayDate = new Date(tY, tM - 1, tD);
   130→  const diffDays = Math.round((todayDate.getTime() - entryDate.getTime()) / 86400000);
   131→
   132→  if (diffDays === 1) return 'Ayer';
   133→  if (diffDays < 7) return 'Esta semana';
   134→  if (diffDays < 30) return 'Este mes';
   135→  return 'Anterior';
   136→}

   155→function getProgressColor(rem: number, limit: number): string {
   156→  if (!isFinite(limit)) return '#c8a55a';
   157→  const pct = rem / limit;
   158→  if (pct > 0.5) return '#c8a55a';
   159→  if (pct > 0.25) return '#e8a849';
   160→  return '#ef4444';
   161→}

   535→  const sendMessage = async () => {

   544→    sendingRef.current = true;
   545→
   546→    const userMessage: Message = {
   547→      id: 'temp-' + Date.now(),
   548→      role: 'user',
   549→      content,
   550→      createdAt: new Date().toISOString(),
   551→    };
   552→    setMessages(prev => [...prev, userMessage]);
   553→    setInput('');
   554→    setSending(true);
   555→
   556→    try {
   557→      const res = await apiFetch('/api/ai/chat', {
   558→        method: 'POST',
   559→        body: JSON.stringify({ threadId: sentThreadId, content }),
   560→      });
   561→
   562→      if (res.status === 403) {
   563→        const data = await res.json();
