   105→interface ParsedCapture {
   106→  description: string;
   107→  amount: number;
   108→  category: string;
   109→  type: 'expense' | 'income';

   111→
   112→function parseQuickCapture(input: string, historyMap: Record<string, string>): ParsedCapture | null {
   113→  if (!input.trim()) return null;
   114→
   115→  const tokens = input.trim().split(/\s+/);
   116→  let amount = 0;
   117→  let descriptionTokens: string[] = [];
   118→  let amountFound = false;
   119→
   120→  for (let i = tokens.length - 1; i >= 0; i--) {
   121→    if (/^[\d.,]+$/.test(tokens[i])) {
   122→      const parsed = parseEuropeanQuick(tokens[i]);
   123→      if (parsed > 0) {
   124→        amount = parsed;
   125→        descriptionTokens = tokens.slice(0, i);
   126→        amountFound = true;
   127→        break;
   128→      }
   129→    }
   130→  }
   131→
   132→  if (!amountFound && /^[\d.,]+$/.test(tokens[0])) {
   133→    const parsed = parseEuropeanQuick(tokens[0]);
   134→    if (parsed > 0) {
   135→      amount = parsed;
   136→      descriptionTokens = tokens.slice(1);
   137→      amountFound = true;
   138→    }
   139→  }
   140→
   141→  if (!amountFound) {
   142→    const match = input.match(/([\d.,]+)\s*€?\s*$/);
   143→    if (match) {
   144→      const parsed = parseEuropeanQuick(match[1]);
   145→      if (parsed > 0) {
   146→        amount = parsed;
   147→        descriptionTokens = [input.replace(match[0], '').trim()];
   148→        amountFound = true;
   149→      }
   150→    }
   151→  }
   152→
   153→  if (!amountFound || amount <= 0) return null;
   154→
   155→  const description = descriptionTokens.join(' ').trim();
   156→  const textToMatch = description || input;
   157→
   158→  let category = '';
   159→  const descLower = textToMatch.toLowerCase();
   160→  if (historyMap[descLower]) {
   161→    category = historyMap[descLower];
   162→  }
   163→
   164→  if (!category) {
   165→    for (const [regex, cat] of KEYWORD_CATEGORIES) {
   166→      if (regex.test(textToMatch)) {
   167→        category = cat;
   168→        break;
   169→      }
   170→    }
   171→  }
   172→
   173→  if (!category) category = 'Otros';
   174→
   175→  const type: 'expense' | 'income' = INCOME_KEYWORDS.test(textToMatch) ? 'income' : 'expense';
   176→
   177→  return {
   178→    description: description || category,
   179→    amount,
   180→    category,
   181→    type,
   182→  };
   183→}
   184→
   185→function parseEuropeanQuick(raw: string): number {
   186→  const trimmed = raw.trim();
   187→  if (!trimmed) return 0;
   188→  const hasComma = trimmed.includes(',');
   189→  const hasDot = trimmed.includes('.');
   190→  let normalized: string;
   191→
   192→  if (hasComma && hasDot) {
   193→    normalized = trimmed.replace(/\./g, '').replace(',', '.');
   194→  } else if (hasComma) {
   195→    normalized = trimmed.replace(',', '.');
   196→  } else if (hasDot) {
   197→    normalized = /^\d{1,3}(\.\d{3})+$/.test(trimmed)
   198→      ? trimmed.replace(/\./g, '')
   199→      : trimmed;
   200→  } else {
   201→    normalized = trimmed;
   202→  }
   203→
   204→  normalized = normalized.replace(/\.$/, '');
   205→  const parsed = parseFloat(normalized);
   206→  return isNaN(parsed) ? 0 : parsed;
   207→}

   213→function getMonthRange(date: Date) {

   218→function filterLogsByRange(logs: FinanceLog[], start: Date, end: Date) {
   219→  return logs.filter(l => {
   220→    const d = new Date(l.date);
   221→    return d >= start && d <= end;
   222→  });
   223→}

   225→const MONTH_NAMES_ES = ['ene', 'feb', 'mar', 'abr', 'may', 'jun', 'jul', 'ago', 'sep', 'oct', 'nov', 'dic'];
   226→const MONTH_LABELS_ES = ['enero', 'febrero', 'marzo', 'abril', 'mayo', 'junio', 'julio', 'agosto', 'septiembre', 'octubre', 'noviembre', 'diciembre'];
   227→const DAY_NAMES_ES = ['dom', 'lun', 'mar', 'mié', 'jue', 'vie', 'sáb'];
   228→
   229→function formatDateLabel(dateStr: string): string {
   230→  const d = new Date(dateStr);
   231→  const day = DAY_NAMES_ES[d.getDay()];
   232→  const date = d.getDate();
   233→  const month = MONTH_NAMES_ES[d.getMonth()];
   234→  return `${day}, ${date} ${month}`;
   235→}

   237→function groupLogsByDate(logs: FinanceLog[]) {
   238→  const groups: Record<string, FinanceLog[]> = {};
   239→  for (const log of logs) {
   240→    const key = formatDateLabel(log.date);
   241→    if (!groups[key]) groups[key] = [];
   242→    groups[key].push(log);
   243→  }
   244→  return groups;
   245→}

   318→function IntentionBalance({ flows, totalExpense }: { flows: IntentionFlow[]; totalExpense: number }) {

   370→function QuickCapture({
   371→  onCapture,
   372→  onFullForm,
   373→  submitting,
   374→  historyMap,
   375→}: {
   376→  onCapture: (data: { type: string; category: string; amount: number; description: string; contexto: string }) => void;
   377→  onFullForm: () => void;
   378→  submitting: boolean;
   379→  historyMap: Record<string, string>;
   380→}) {
   381→  const [input, setInput] = useState('');
   382→  const [contexto, setContexto] = useState('');
   383→  const parsed = useMemo(() => parseQuickCapture(input, historyMap), [input, historyMap]);
   384→
   385→  const handleSubmit = () => {
   386→    if (!parsed || parsed.amount <= 0 || submitting) return;
   387→    onCapture({
   388→      type: parsed.type,
   389→      category: parsed.category,
   390→      amount: parsed.amount,
   391→      description: parsed.description,
   392→      contexto: contexto.trim(),
   393→    });
   394→    setInput('');
   395→    setContexto('');
   396→  };
   397→
   398→  const handleKeyDown = (e: React.KeyboardEvent) => {
   399→    if (e.key === 'Enter' && parsed && parsed.amount > 0) {
   400→      e.preventDefault();
   401→      handleSubmit();
   402→    }
   403→  };
   404→
   405→  return (
   406→    <div>
   407→      <div className="relative">
   408→        <input
   409→          type="text"
   410→          value={input}
   411→          onChange={(e) => setInput(e.target.value)}
   412→          onKeyDown={handleKeyDown}
   413→          placeholder="Café 3,50 · Taxi 14 · Nómina 1250"
   414→          className="w-full bg-[#000000] border border-[#1a1a1a] rounded-xl px-4 py-3.5 text-white text-base placeholder-[#444] focus:outline-none focus:border-champagne/40 transition-colors"
   415→          autoFocus
   416→          inputMode="text"
   417→          autoComplete="off"
   418→        />
   419→      </div>
   420→
   421→      {parsed && parsed.amount > 0 ? (
   422→        <div className="mt-2 flex items-center justify-between px-0.5">
   423→          <div className="flex items-center gap-1.5">
   424→            <span className={`text-[11px] font-medium ${parsed.type === 'income' ? 'text-champagne' : 'text-red-400/70'}`}>
   425→              {parsed.type === 'income' ? 'Ingreso' : 'Gasto'}
   426→            </span>
   427→            <span className="text-[#222] text-[11px]">·</span>
   428→            <span className="text-[11px] text-[#666]">{parsed.category}</span>
   429→            <span className="text-[#222] text-[11px]">·</span>
   430→            <span className={`text-[11px] font-medium tabular-nums ${parsed.type === 'income' ? 'text-champagne' : 'text-red-400/70'}`}>
   431→              {parsed.type === 'income' ? '+' : '-'}{formatCurrency(parsed.amount)}
   432→            </span>
   433→          </div>
   434→          <button
   435→            onClick={handleSubmit}
   436→            disabled={submitting}
   437→            className="bg-champagne text-black px-3 py-1 rounded-lg text-xs font-semibold hover:bg-champagne-hover active:scale-95 transition-all disabled:opacity-50"
   438→          >
   439→            {submitting ? '...' : 'OK'}
   440→          </button>
   441→        </div>
   442→      ) : (
   443→        <p className="mt-2 text-[10px] text-[#333] px-0.5">Escribe concepto y cantidad</p>
   444→      )}
   445→
   446→      <div className="mt-3">
   447→        <input
   448→          type="text"
   449→          value={contexto}
   450→          onChange={(e) => setContexto(e.target.value)}
   451→          placeholder="¿Qué pasó? (opcional)"
   452→          className="w-full bg-transparent border-b border-[#1a1a1a] px-0 py-1.5 text-[12px] text-[#888] placeholder-[#333] focus:outline-none focus:border-champagne/25 transition-colors italic"
   453→          autoComplete="off"
   454→        />
   455→      </div>
   456→
   457→      <button
   458→        onClick={onFullForm}
   459→        className="mt-2 text-[10px] text-[#444] hover:text-[#666] transition-colors tracking-wide"
   460→      >
   461→        Formulario completo
   462→      </button>
   463→    </div>
   464→  );
   465→}
