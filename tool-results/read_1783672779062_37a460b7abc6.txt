    15→

    21→  sleep: number;
    22→  stress: number;
    23→  notes: string | null;
    24→  createdAt: string;
    25→}
    26→
    27→interface NutritionLog {
    28→  id: string;
    29→  date: string;
    30→  meals: string | null;
    31→  water: number;
    32→  calories: number | null;
    33→  notes: string | null;
    34→  createdAt: string;
    35→}
    36→
    37→// Rating input — module-level component to prevent remount flicker.
    38→// Previously defined inside EnergiaPage, causing React to unmount+remount
    39→// all rating buttons on every state change (visual flicker + lost :active state).
    40→function RatingInput({ value, onChange, label }: { value: number; onChange: (v: number) => void; label: string }) {
    41→  return (
    42→    <div>
    43→      <label className="text-sm text-[#999] mb-1 block">{label}</label>
    44→      <div className="flex gap-2">
    45→        {[1, 2, 3, 4, 5].map((n) => (
    46→          <button key={n} onClick={() => onChange(n)}
    47→            className={`rating-btn w-10 h-10 rounded-lg border text-sm font-medium transition-colors ${n <= value ? 'bg-champagne border-champagne text-black' : 'bg-[#000000] border-[#1a1a1a] text-[#666] hover:border-champagne'}`}>
    48→            {n}
    49→          </button>
    50→        ))}
    51→      </div>
    52→    </div>
    53→  );
    54→}
    55→
    56→export default function EnergiaPage() {
    57→  const { apiFetch } = useApi();
    58→  const { user } = useAuth();
    59→  const [wellnessLogs, setWellnessLogs] = useState<WellnessLog[]>([]);
    60→  const [nutrition, setNutrition] = useState<NutritionLog[]>([]);
    61→  const [showWellness, setShowWellness] = useState(false);
    62→  const [showNutrition, setShowNutrition] = useState(false);
    63→  const [wellnessForm, setWellnessForm] = useState({ mood: 3, energy: 3, sleep: 3, stress: 3, notes: '' });
    64→  const [nutritionForm, setNutritionForm] = useState({ meals: '', water: 0, calories: 0, notes: '' });
    65→  const [loading, setLoading] = useState(true);
    66→  const [editingWellness, setEditingWellness] = useState<WellnessLog | null>(null);
    67→  const [editWellnessForm, setEditWellnessForm] = useState({ mood: 3, energy: 3, sleep: 3, stress: 3, notes: '' });
    68→  const [editWellnessSaving, setEditWellnessSaving] = useState(false);
    69→  const [editingNutrition, setEditingNutrition] = useState<NutritionLog | null>(null);
    70→  const [editNutritionForm, setEditNutritionForm] = useState({ meals: '', water: 0, calories: 0, notes: '' });
    71→  const [editNutritionSaving, setEditNutritionSaving] = useState(false);
    72→  const [pendingDeleteId, setPendingDeleteId] = useState<{ id: string; type: 'wellness' | 'nutrition' } | null>(null);
    73→  const [fetchError, setFetchError] = useState(false);
    74→  const [submittingWellness, setSubmittingWellness] = useState(false);
    75→  const [submittingNutrition, setSubmittingNutrition] = useState(false);
    76→
    77→  // Lock body scroll when modal is open — save/restore scroll position
    78→  useEffect(() => {
    79→    if (editingWellness || editingNutrition || pendingDeleteId) {
    80→      document.body.classList.add('scroll-locked');
    81→      return () => {
    82→        document.body.classList.remove('scroll-locked');
    83→      };
    84→    }
    85→  }, [editingWellness, editingNutrition, pendingDeleteId]);
    86→
    87→  const fetchData = useCallback(async () => {
    88→    setLoading(true);
    89→    setFetchError(false);
    90→    try {
    91→      const [wRes, nRes] = await Promise.all([
    92→        apiFetch('/api/wellness'),
    93→        apiFetch('/api/nutrition'),
    94→      ]);
    95→      if (wRes.ok) { const d = await wRes.json(); setWellnessLogs(d.logs); }
    96→      if (nRes.ok) { const d = await nRes.json(); setNutrition(d.logs); }

    98→      console.error(e);
    99→      setFetchError(true);
   100→    } finally { setLoading(false); }
   101→  }, [apiFetch]);
   102→
   103→  useEffect(() => {
   104→    fetchData();
   105→  }, [fetchData]);
   106→
   107→  const submitWellness = async () => {
   108→    if (submittingWellness) return;
   109→    setSubmittingWellness(true);
   110→    try {
   111→      const res = await apiFetch('/api/wellness', {
   112→        method: 'POST',
   113→        body: JSON.stringify({ date: getTodayDateKey(), ...wellnessForm }),
   114→      });
   115→      if (res.ok) {
   116→        const data = await res.json();
   117→        setWellnessLogs(prev => {
   118→          const filtered = prev.filter(l => l.id !== data.log.id);
   119→          return [data.log, ...filtered];
   120→        });
   121→        setWellnessForm({ mood: 3, energy: 3, sleep: 3, stress: 3, notes: '' });
   122→        setShowWellness(false);
   123→      }

   125→    finally { setSubmittingWellness(false); }
   126→  };
   127→
   128→  const submitNutrition = async () => {
   129→    if (submittingNutrition) return;
   130→    setSubmittingNutrition(true);
   131→    try {
   132→      const res = await apiFetch('/api/nutrition', {
   133→        method: 'POST',
   134→        body: JSON.stringify({ date: getTodayDateKey(), ...nutritionForm }),
   135→      });
   136→      if (res.ok) {
   137→        const data = await res.json();
   138→        setNutrition(prev => {
   139→          const filtered = prev.filter(l => l.id !== data.log.id);
   140→          return [data.log, ...filtered];
   141→        });
   142→        setNutritionForm({ meals: '', water: 0, calories: 0, notes: '' });
   143→        setShowNutrition(false);
   144→      }
   145→    } catch (error) { console.error('Error submitting nutrition:', error); }
   146→    finally { setSubmittingNutrition(false); }
   147→  };

   149→  const startEditWellness = (log: WellnessLog) => {
   150→    setEditingWellness(log);
   151→    setEditWellnessForm({ mood: log.mood, energy: log.energy, sleep: log.sleep, stress: log.stress, notes: log.notes || '' });
   152→  };

   154→  const saveEditWellness = async () => {
   155→    if (!editingWellness) return;
   156→    setEditWellnessSaving(true);

   158→      const res = await apiFetch('/api/wellness', {
   159→        method: 'PUT',
   160→        body: JSON.stringify({ logId: editingWellness.id, ...editWellnessForm }),
   161→      });
   162→      if (res.ok) {
   163→        const data = await res.json();
   164→        setWellnessLogs(prev => prev.map(l => l.id === editingWellness.id ? data.log : l));
   165→        setEditingWellness(null);
   166→      } else {
   167→        const errData = await res.json().catch(() => ({}));
   168→        console.error('Wellness PUT failed:', res.status, errData);
   169→      }
   170→    } catch (error) { console.error('Error updating wellness log:', error); }
   171→    finally { setEditWellnessSaving(false); }
   172→  };
   173→
   174→  const startEditNutrition = (log: NutritionLog) => {
   175→    setEditingNutrition(log);
   176→    setEditNutritionForm({ meals: log.meals || '', water: log.water, calories: log.calories || 0, notes: log.notes || '' });
   177→  };
   178→
   179→  const saveEditNutrition = async () => {
   180→    if (!editingNutrition) return;
   181→    setEditNutritionSaving(true);
   182→    try {
   183→      const res = await apiFetch('/api/nutrition', {
   184→        method: 'PUT',
   185→        body: JSON.stringify({ logId: editingNutrition.id, ...editNutritionForm }),
   186→      });
   187→      if (res.ok) {
   188→        const data = await res.json();
   189→        setNutrition(prev => prev.map(l => l.id === editingNutrition.id ? data.log : l));
   190→        setEditingNutrition(null);

   192→        const errData = await res.json().catch(() => ({}));
   193→        console.error('Nutrition PUT failed:', res.status, errData);
   194→      }
   195→    } catch (error) { console.error('Error updating nutrition log:', error); }
   196→    finally { setEditNutritionSaving(false); }

   198→
   199→  const confirmDelete = async () => {

   209→        if (pendingDeleteId.type === 'wellness') {
   210→          setWellnessLogs(prev => prev.filter(l => l.id !== pendingDeleteId.id));

   212→          setNutrition(prev => prev.filter(l => l.id !== pendingDeleteId.id));
   213→        }
   214→      } else {
   215→        const errData = await res.json().catch(() => ({}));
   216→        console.error('DELETE failed:', res.status, errData);
   217→      }
   218→    } catch (error) { console.error('Error deleting log:', error); }
   219→    finally { setPendingDeleteId(null); }

   222→  if (loading) {

   229→        <PremiumErrorState
   230→          variant="loading"
   231→          title="No se pudo cargar el imperio"
   232→          onRetry={fetchData}
   233→          size="md"
   234→        />
   235→      </div>
   236→    );
   237→  }
   238→
   239→  return (
   240→    <div className="max-w-4xl mx-auto space-y-6 sm:space-y-8">

   243→        <div className="fixed inset-0 z-50 flex items-center justify-center modal-backdrop p-4" onClick={() => setEditingWellness(null)}>
   244→          <div className="modal-content p-8 max-w-md w-full" onClick={(e) => e.stopPropagation()}>
   245→            <div className="w-12 h-12 rounded-full bg-champagne/10 flex items-center justify-center mx-auto mb-5">
   246→              <Pencil size={20} className="text-champagne" />
   247→            </div>
   248→            <h3 className="text-lg font-bold text-white text-center mb-6">Editar bienestar</h3>
   249→            <div className="space-y-4">
   250→              <RatingInput label="Estado de ánimo" value={editWellnessForm.mood} onChange={(v) => setEditWellnessForm({ ...editWellnessForm, mood: v })} />
   251→              <RatingInput label="Energía" value={editWellnessForm.energy} onChange={(v) => setEditWellnessForm({ ...editWellnessForm, energy: v })} />
   252→              <RatingInput label="Sueño" value={editWellnessForm.sleep} onChange={(v) => setEditWellnessForm({ ...editWellnessForm, sleep: v })} />
   253→              <RatingInput label="Estrés" value={editWellnessForm.stress} onChange={(v) => setEditWellnessForm({ ...editWellnessForm, stress: v })} />
   254→              <textarea placeholder="Notas (opcional)" value={editWellnessForm.notes} onChange={(e) => setEditWellnessForm({ ...editWellnessForm, notes: e.target.value })}
   255→                className="w-full bg-[#000000] border border-[#1a1a1a] rounded-lg px-4 py-3 text-white text-base sm:text-sm focus:outline-none focus:border-champagne/50 transition-colors placeholder-[#666] h-20 resize-none" />
   256→            </div>
   257→            <div className="flex items-center justify-center gap-3 mt-7">
   258→              <button onClick={() => setEditingWellness(null)} className="bg-[#000000] border border-[#333] text-[#999] font-medium px-5 py-2.5 rounded-xl hover:bg-[#111] transition-colors">Cancelar</button>
   259→              <button onClick={saveEditWellness} disabled={editWellnessSaving} className="bg-champagne text-black font-semibold px-5 py-2.5 rounded-xl hover:bg-champagne-hover transition-colors disabled:opacity-50 disabled:cursor-not-allowed">{editWellnessSaving ? 'Guardando...' : 'Guardar'}</button>
   260→            </div>
   261→          </div>
   262→        </div>
   263→      )}
   264→
   265→      {/* Edit Nutrition Overlay */}
   266→      {editingNutrition && (
   267→        <div className="fixed inset-0 z-50 flex items-center justify-center modal-backdrop p-4" onClick={() => setEditingNutrition(null)}>
   268→          <div className="modal-content p-8 max-w-md w-full" onClick={(e) => e.stopPropagation()}>
   269→            <div className="w-12 h-12 rounded-full bg-champagne/10 flex items-center justify-center mx-auto mb-5">
   270→              <Pencil size={20} className="text-champagne" />
   271→            </div>
   272→            <h3 className="text-lg font-bold text-white text-center mb-6">Editar nutrición</h3>
   273→            <div className="space-y-3">

   275→                className="w-full bg-[#000000] border border-[#1a1a1a] rounded-lg px-4 py-3 text-white text-base sm:text-sm focus:outline-none focus:border-champagne/50 transition-colors placeholder-[#666] h-20 resize-none" />
   276→              <div className="grid grid-cols-2 gap-3">
   277→                <div>
   278→                  <label className="text-xs text-[#999] uppercase tracking-wider font-medium mb-2 block">Vasos de agua</label>
   279→                  <NumericInput value={editNutritionForm.water} onChange={(v) => setEditNutritionForm({ ...editNutritionForm, water: v })} inputMode="numeric" allowDecimal={false} min={0}
   280→                    className="w-full bg-[#000000] border border-[#1a1a1a] rounded-lg px-4 py-3 text-white text-base sm:text-sm focus:outline-none focus:border-champagne/50 transition-colors" />
   281→                </div>

   285→                    className="w-full bg-[#000000] border border-[#1a1a1a] rounded-lg px-4 py-3 text-white text-base sm:text-sm focus:outline-none focus:border-champagne/50 transition-colors" />
   286→                </div>
   287→              </div>
   288→            </div>
   289→            <div className="flex items-center justify-center gap-3 mt-7">
   290→              <button onClick={() => setEditingNutrition(null)} className="bg-[#000000] border border-[#333] text-[#999] font-medium px-5 py-2.5 rounded-xl hover:bg-[#111] transition-colors">Cancelar</button>
   291→              <button onClick={saveEditNutrition} disabled={editNutritionSaving} className="bg-champagne text-black font-semibold px-5 py-2.5 rounded-xl hover:bg-champagne-hover transition-colors disabled:opacity-50 disabled:cursor-not-allowed">{editNutritionSaving ? 'Guardando...' : 'Guardar'}</button>
   292→            </div>
   293→          </div>
   294→        </div>
   295→      )}
   296→
   297→      {/* Delete Confirmation Overlay */}
   298→      {pendingDeleteId && (
   299→        <div className="fixed inset-0 z-50 flex items-center justify-center modal-backdrop p-4" onClick={() => setPendingDeleteId(null)}>
   300→          <div className="modal-content-destructive p-8 max-w-md w-full" onClick={(e) => e.stopPropagation()}>
   301→            <div className="w-12 h-12 rounded-full bg-red-500/10 flex items-center justify-center mx-auto mb-4">
   302→              <Trash2 size={22} className="text-red-400" />
   303→            </div>
   304→            <h3 className="text-lg font-bold text-white mb-2">Eliminar registro</h3>
   305→            <p className="text-[#999] text-sm mb-6">Esta acción no se puede deshacer</p>
   306→            <div className="flex items-center justify-center gap-3">
   307→              <button onClick={() => setPendingDeleteId(null)} className="bg-[#000000] border border-[#333] text-[#999] font-medium px-5 py-2.5 rounded-xl hover:bg-[#111] transition-colors">Cancelar</button>
   308→              <button onClick={confirmDelete} className="bg-red-500/90 text-white font-medium px-5 py-2.5 rounded-xl hover:bg-red-500 transition-colors">Eliminar</button>
   309→            </div>
   310→          </div>
   311→        </div>

   313→
   314→      <div className="flex items-center gap-4">
   315→        <div className="w-12 h-12 sm:w-14 sm:h-14 rounded-xl bg-champagne/10 flex items-center justify-center">
   316→          <Zap size={28} className="text-champagne" />
   317→        </div>
   318→        <div>
   319→          <h1 className="title-page">Energía</h1>
   320→          <p className="subtitle-silent mt-1">Cuerpo y vitalidad</p>
   321→        </div>
   322→      </div>
   323→
   324→      {/* Contextual Help — progressive disclosure */}
   325→      <ContextualHelp
   326→        storageKey="vitazen_help_energia"
   327→        title="Energía"
   328→        text="Registra tu bienestar físico y nutrición diaria. Ánimo, energía, sueño y estrés, junto con tus comidas y agua."
   329→      />
   330→
   331→      {/* Wellness Log */}
   332→      <div className="section-enter-1 bg-[#0a0a0a] border border-[#1a1a1a] rounded-xl p-5 sm:p-6">
   333→        <div className="flex items-center justify-between mb-4">
   334→          <h2 className="text-lg font-semibold text-white">Registro de Bienestar</h2>
   335→          <button onClick={() => setShowWellness(!showWellness)} className="touch-press text-sm text-champagne hover:text-champagne-hover">
   336→            + Registrar hoy
   337→          </button>
   338→        </div>
   339→
   340→        {showWellness && (
   341→          <div className="bg-[#000000] border border-[#1a1a1a] rounded-lg p-4 mb-4 space-y-4">

   343→            <RatingInput label="Energía" value={wellnessForm.energy} onChange={(v) => setWellnessForm({ ...wellnessForm, energy: v })} />
   344→            <RatingInput label="Sueño" value={wellnessForm.sleep} onChange={(v) => setWellnessForm({ ...wellnessForm, sleep: v })} />
   345→            <RatingInput label="Estrés" value={wellnessForm.stress} onChange={(v) => setWellnessForm({ ...wellnessForm, stress: v })} />

   348→            <div className="flex gap-2">
   349→              <button onClick={submitWellness} disabled={submittingWellness} className="touch-press bg-champagne text-black font-semibold px-4 py-2 rounded-xl text-sm hover:bg-champagne-hover disabled:opacity-50 disabled:cursor-not-allowed">{submittingWellness ?    350→              <button onClick={() => setShowWellness(false)} className="touch-press text-[#999] px-4 py-2 text-sm">Cancelar</button>

   356→            {wellnessLogs.slice(0, 7).map((log) => (
   357→              <div key={log.id} className="bg-[#000000] border border-[#1a1a1a] rounded-lg p-3 sm:p-4 group hover:border-[#222] transition-colors">
   358→                <div className="flex items-center justify-between">
   359→                  <div className="flex items-center gap-3 sm:gap-4">
   360→                    <div className="w-10 h-10 rounded-lg bg-champagne/10 flex items-center justify-center shrink-0">
   361→                      <Heart size={16} className="text-champagne" />
   362→                    </div>
   363→                    <div>
   364→                      <div className="flex gap-1.5 sm:gap-2 flex-wrap text-xs mb-1">
   365→                        <span className="text-champagne">Ánimo: <PrivacyMask compact>{log.mood}</PrivacyMask></span>

   367→                        <span className="text-champagne">Sueño: <PrivacyMask compact>{log.sleep}</PrivacyMask></span>
   368→                      </div>
   369→                      <div className="flex items-center gap-3">
   370→                        <span className="flex items-center gap-1 text-xs text-[#999]"><Calendar size={11} />{new Date(log.date).toLocaleDateString('es', { day: 'numeric', month: 'short', year: 'numeric' })}</span>

   378→                    <button onClick={() => setPendingDeleteId({ id: log.id, type: 'wellness' })} className="p-2.5 rounded-lg hover:bg-red-500/10 text-[#666] hover:text-red-400 transition-all touch-press" title="Eliminar"><Trash2 size={14} /></button>
