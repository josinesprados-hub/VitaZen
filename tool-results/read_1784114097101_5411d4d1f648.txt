    89→function getRelativeDate(dateStr: string): string {
    90→  const date = new Date(dateStr);
    91→  const now = new Date();
    92→  const diffMs = now.getTime() - date.getTime();
    93→  const diffMins = Math.floor(diffMs / 60000);
    94→  const diffHours = Math.floor(diffMs / 3600000);
    95→
    96→  // Calendar-day difference using Madrid-normalized dates
    97→  const entryKey = getMadridDateKey(date);
    98→  const todayKey = getTodayDateKey();
    99→  const [eY, eM, eD] = entryKey.split('-').map(Number);
   100→  const [tY, tM, tD] = todayKey.split('-').map(Number);
   101→  const entryDate = new Date(eY, eM - 1, eD);
   102→  const todayDate = new Date(tY, tM - 1, tD);
   103→  const diffDays = Math.round((todayDate.getTime() - entryDate.getTime()) / 86400000);
   104→
   105→  if (diffMins < 1 && diffDays === 0) return 'Ahora';
   106→  if (diffMins < 60 && diffDays === 0) return `Hace ${diffMins} min`;
   107→  if (diffHours < 24 && diffDays === 0) return `Hace ${diffHours}h`;
   108→  if (diffDays === 1) return 'Ayer';
   109→  if (diffDays < 7) return `Hace ${diffDays} días`;
   110→  if (diffDays < 30) {
   111→    const weeks = Math.floor(diffDays / 7);
   112→    return weeks === 1 ? 'Hace 1 semana' : `Hace ${weeks} semanas`;
   113→  }
   114→  return entryDate.toLocaleDateString('es-ES', { day: 'numeric', month: 'short' });
   115→}

   117→function getDateGroup(dateStr: string): string {
   118→  // Calendar-day difference using Madrid-normalized dates
   119→  const entryKey = getMadridDateKey(new Date(dateStr));
   120→  const todayKey = getTodayDateKey();
   121→
   122→  if (entryKey === todayKey) return 'Hoy';
   123→
   124→  const [eY, eM, eD] = entryKey.split('-').map(Number);
   125→  const [tY, tM, tD] = todayKey.split('-').map(Number);
   126→  const entryDate = new Date(eY, eM - 1, eD);
   127→  const todayDate = new Date(tY, tM - 1, tD);
   128→  const diffDays = Math.round((todayDate.getTime() - entryDate.getTime()) / 86400000);
   129→
   130→  if (diffDays === 1) return 'Ayer';
   131→  if (diffDays < 7) return 'Esta semana';
   132→  if (diffDays < 30) return 'Este mes';
   133→  return 'Anterior';
   134→}

   154→  if (!isFinite(limit)) return '#c8a55a';
   155→  const pct = rem / limit;
   156→  if (pct > 0.5) return '#c8a55a';
   157→  if (pct > 0.25) return '#e8a849';

   528→  const sendMessage = async () => {

   534→    lastSendTime.current = now;
   535→
   536→    const sentThreadId = activeThread;
   537→    sendingRef.current = true;
   538→
   539→    const userMessage: Message = {
   540→      id: 'temp-' + Date.now(),
   541→      role: 'user',
   542→      content,
   543→      createdAt: new Date().toISOString(),
   544→    };
   545→    setMessages(prev => [...prev, userMessage]);
   546→    setInput('');
   547→    setSending(true);
   548→
   549→    try {
   550→      const res = await apiFetch('/api/ai/chat', {
   551→        method: 'POST',
   552→        body: JSON.stringify({ threadId: sentThreadId, content }),
   553→      });
   554→
   555→      if (res.status === 403) {
   556→        const data = await res.json();
   557→        setShowLimitModal(true);
   558→        setRemaining(0);
   559→        setDailyLimit(data.limit || 15);
   560→        // Remove the optimistic user message since it was blocked
   561→        setMessages(prev => prev.filter(m => m.id !== userMessage.id));
   562→        // L-4 FIX: Restore the input text so the user doesn't lose their message.
   563→        // Previously the input was cleared (line 547) and never restored on 403.
   564→        setInput(content);
   565→        return;
   566→      }
