   162→/* ─── Breathing Pattern Visualizer ─── */
   163→function PatternFlow({ phases }: { phases: BreathingPhase[] }) {
   164→  const activePhases = phases.filter(p => p.seconds > 0);
   165→  const isNatural = activePhases.length === 0;
   166→
   167→  if (isNatural) {
   168→    return (
   169→      <div className="flex items-center justify-center gap-3 py-3">
   170→        <div className="w-12 h-12 rounded-full border border-champagne/30 flex items-center justify-center">
   171→          <Wind size={18} className="text-champagne" />
   172→        </div>
   173→        <div>
   174→          <p className="text-champagne text-sm font-medium">Ritmo natural</p>
   175→          <p className="text-[#666] text-xs">Sin patrón fijo</p>
   176→        </div>
   177→      </div>
   178→    );
   179→  }
   180→
   181→  return (
   182→    <div className="flex items-center justify-center gap-1.5 py-3 overflow-x-auto">
   183→      {activePhases.map((phase, i) => (
   184→        <div key={i} className="flex items-center gap-1.5 shrink-0">
   185→          <div className="flex flex-col items-center">
   186→            <div className={`w-10 h-10 rounded-full border flex items-center justify-center ${
   187→              phase.label.includes('Mantén')
   188→                ? 'border-champagne/35 bg-champagne/8'
   189→                : 'border-champagne/50 bg-champagne/12'
   190→            }`}>
   191→              <span className="text-champagne text-xs font-semibold">{phase.seconds}s</span>
   192→            </div>
   193→            <span className="text-[10px] text-[#666] mt-1.5 whitespace-nowrap">{phase.label}</span>
   194→          </div>
   195→          {i < activePhases.length - 1 && (
   196→            <div className="w-4 h-px bg-[#333] mb-5 shrink-0" />
   197→          )}
   198→        </div>
   199→      ))}
   200→    </div>
   201→  );
   202→}

   215→  const [pendingDeleteId, setPendingDeleteId] = useState<string | null>(null);
   216→  const [editingSession, setEditingSession] = useState<Meditation | null>(null);
   217→  const [editDuration, setEditDuration] = useState<number>(0);
   218→  const [editType, setEditType] = useState<string>('');
   219→  const [editSaving, setEditSaving] = useState(false);
   220→  const [fetchError, setFetchError] = useState(false);
   221→
   222→  // Ref for auto-scrolling into view when meditation starts
   223→  const breathingSectionRef = useRef<HTMLDivElement>(null);
   224→
   225→  // Lock body scroll when any modal is open — save/restore scroll position
   226→  useEffect(() => {
   227→    const anyOverlayOpen = !!(pendingDeleteId || completedSession || editingSession);
   228→    if (anyOverlayOpen) {
   229→      document.body.classList.add('scroll-locked');
   230→      return () => {
   231→        document.body.classList.remove('scroll-locked');
   232→      };
   233→    }
   234→  }, [pendingDeleteId, completedSession, editingSession]);
   235→
   236→  const fetchData = useCallback(async () => {
   237→    setLoading(true);
   238→    setFetchError(false);
   239→    try {
   240→      const [medRes] = await Promise.all([
   241→        apiFetch('/api/meditation'),
   242→      ]);
   243→      if (medRes.ok) { const d = await medRes.json(); setSessions(d.sessions); }
   244→    } catch (error) {
   245→      console.error('Error:', error);
   246→      setFetchError(true);
   247→    } finally {
   248→      setLoading(false);
   249→    }
   250→  }, [apiFetch]);
   251→
   252→  useEffect(() => {
   253→    fetchData();
   254→  }, [fetchData]);
   255→
   256→  useEffect(() => {
   257→    let interval: NodeJS.Timeout;
   258→    if (meditating && !paused) {
   259→      interval = setInterval(() => setTimer(t => t + 1), 1000);
   260→    }
   261→    return () => clearInterval(interval);
   262→  }, [meditating, paused]);
   263→
   264→  const startMeditation = (type: BreathingTechnique) => {
   265→    setSelectedType(type);
   266→    setTimer(0);
   267→    setPaused(false);
   268→    setMeditating(true);
   269→    // Scroll to breathing section after render — requestAnimationFrame ensures
   270→    // the DOM has updated (guide unmounted, timer mounted) before scrolling
   271→    requestAnimationFrame(() => {
   272→      breathingSectionRef.current?.scrollIntoView({ behavior: 'smooth', block: 'start' });
   273→    });
   274→  };
   275→
   276→  const endMeditation = async () => {
   277→    setMeditating(false);
   278→    setPaused(false);
   279→    const duration = Math.max(1, Math.floor(timer / 60));
   280→    const type = selectedType.type;
   281→    setCompletedSession({ duration, type });
   282→    try {
   283→      const res = await apiFetch('/api/meditation', {
   284→        method: 'POST',
   285→        body: JSON.stringify({ duration, type }),
   286→      });
   287→      if (res.ok) {
   288→        const data = await res.json();
   289→        setSessions(prev => [data.session, ...prev]);
   290→      }
   291→    } catch (error) {
   292→      console.error('Error saving meditation:', error);
   293→    }
   294→    setTimer(0);
   295→  };
   296→
   297→  const formatTime = (seconds: number) => {
   298→    const m = Math.floor(seconds / 60);
   299→    const s = seconds % 60;
   300→    return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
   301→  };
   302→
   303→  const startEdit = (session: Meditation) => {
   304→    setEditingSession(session);
   305→    setEditDuration(session.duration);
   306→    setEditType(session.type);
   307→  };
   308→
   309→  const saveEdit = async () => {
   310→    if (!editingSession) return;
   311→    setEditSaving(true);
   312→    try {
   313→      const res = await apiFetch('/api/meditation', {
   314→        method: 'PUT',
   315→        body: JSON.stringify({ sessionId: editingSession.id, duration: editDuration, type: editType }),
   316→      });
   317→      if (res.ok) {
   318→        const data = await res.json();
   319→        setSessions(prev => prev.map(s => s.id === editingSession.id ? data.session : s));
   320→        setEditingSession(null);
   321→      } else {
   322→        const errData = await res.json().catch(() => ({}));
   323→        console.error('Meditation PUT failed:', res.status, errData);
   324→      }
   325→    } catch (error) {
   326→      console.error('Error updating session:', error);
   327→    } finally {
   328→      setEditSaving(false);
   329→    }
   330→  };
   331→
   332→  const confirmDelete = async () => {
   333→    if (!pendingDeleteId) return;
   334→    try {
   335→      const res = await apiFetch('/api/meditation', {
   336→        method: 'DELETE',
   337→        body: JSON.stringify({ sessionId: pendingDeleteId }),
   338→      });
   339→      if (res.ok) {
   340→        setSessions(prev => prev.filter(s => s.id !== pendingDeleteId));
   341→      } else {
   342→        const errData = await res.json().catch(() => ({}));
   343→        console.error('Meditation DELETE failed:', res.status, errData);
   344→      }
   345→    } catch (error) {
   346→      console.error('Error deleting session:', error);
   347→    } finally {
   348→      setPendingDeleteId(null);
   349→    }
   350→  };
   351→
   352→  const guideTechnique = viewingGuide

   368→        />
   369→      </div>
   370→    );
   371→  }
   372→
   373→  return (

   379→          onClick={() => setCompletedSession(null)}
   380→        >
   381→          <div

   383→            onClick={(e) => e.stopPropagation()}
   384→          >
   385→            <div className="w-16 h-16 rounded-full bg-champagne/10 flex items-center justify-center mb-5 shrink-0">
